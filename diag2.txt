    /** @return array<string, mixed> */
    public function analyze(string $path, string $ext, ?int $sheet = null, ?int $headerLine = null): array
    {
        if ($sheet !== null && $headerLine !== null) {
            $sheetIdx = $sheet;
            $line = $headerLine;
            $sheets = $this->detect($path, $ext, $sheet)[2];
        } else {
            [$sheetIdx, $line, $sheets] = $this->detect($path, $ext, $sheet);
        }

        [$headers, $rows, $total] = $this->readSheet($path, $ext, $sheetIdx, $line, 40);
        $columns = $this->suggest($headers, $this->columnValues($headers, $rows));

        $warnings = [];
        $mapped = array_filter(array_column($columns, 'field'));
        if (! in_array('name', $mapped, true) && ! in_array('slug', $mapped, true)) {
            $warnings[] = 'Aucune colonne « nom » reconnue : choisissez-la dans la liste ci-dessous.';
        }
        if (! in_array('price', $mapped, true)) {
            $warnings[] = 'Aucune colonne « prix » : les nouveaux produits ne pourront pas être créés (la mise à jour reste possible).';
        }
        if ($total > self::MAX_ROWS) {
            $warnings[] = 'Le fichier dépasse '.self::MAX_ROWS.' lignes : découpez-le en plusieurs fichiers.';
        }
        if (in_array($ext, ['xlsx'], true) && $this->hasMergedCells($path, $sheetIdx)) {
            $warnings[] = 'Des cellules fusionnées ont été détectées : seule la première ligne d\'un bloc fusionné contient la valeur, les autres paraîtront vides.';
        }
        if (count($sheets) > 1) {
            $name = collect($sheets)->firstWhere('index', $sheetIdx)['name'] ?? '';
            $warnings[] = 'Le fichier contient '.count($sheets).' feuilles. Feuille utilisée : « '.$name.' ».';
        }

        return [
            'sheets' => $sheets,
            'sheet' => $sheetIdx,
            'header_line' => $line,
            'total_rows' => min($total, self::MAX_ROWS + 1),
            'max_rows' => self::MAX_ROWS,
            'columns' => $columns,
            'fields' => collect(self::FIELDS)->map(fn ($label, $key) => ['key' => $key, 'label' => $label])->values()->all(),
            'warnings' => $warnings,
        ];
    }

    /** @return array<string, mixed> */
    public function run(string $path, string $ext, bool $dryRun, bool $publishDefault, ?User $admin, array $opts = []): array
    {
        $this->opts = $opts + [
            'mode' => 'both', 'keep_stock' => false, 'create_categories' => false,
            'mapping' => null, 'sheet' => null, 'header_line' => null, 'remember' => false,
        ];

        if ($this->opts['sheet'] !== null && $this->opts['header_line'] !== null) {
            $sheetIdx = (int) $this->opts['sheet'];
            $headerLine = (int) $this->opts['header_line'];
        } else {
            [$sheetIdx, $headerLine] = $this->detect($path, $ext, $this->opts['sheet'] !== null ? (int) $this->opts['sheet'] : null);
        }

        [$headers, $rows] = $this->readSheet($path, $ext, $sheetIdx, $headerLine, null);

        $warnings = [];
        $map = [];
        if (is_array($this->opts['mapping'])) {
            foreach ($this->opts['mapping'] as $i => $field) {
                $field = (string) $field;
                if ($field === '' || ! isset(self::FIELDS[$field])) {
                    continue;
                }
                if (in_array($field, $map, true)) {
                    $warnings[] = 'Le champ « '.self::FIELDS[$field].' » est choisi pour plusieurs colonnes : seule la première est utilisée.';
                    continue;
                }
                $map[(int) $i] = $field;
            }
        } else {
            foreach ($this->suggest($headers, $this->columnValues($headers, $rows)) as $col) {
                if ($col['field'] !== null) {
                    $map[$col['index']] = $col['field'];
                }
            }
        }

        if (! in_array('name', $map, true) && ! in_array('slug', $map, true)) {
            throw new \RuntimeException('Aucune colonne « nom » associée. Vérifiez la correspondance des colonnes avant d\'importer.');
        }

        $unknown = [];
        foreach ($headers as $i => $h) {
            $label = $this->str($h);
            if ($label !== '' && ! isset($map[$i])) {
                $unknown[] = $label;
            }
        }

        $this->report = [
            'dry_run' => $dryRun, 'total' => 0, 'created' => 0, 'updated' => 0,
            'unchanged' => 0, 'skipped' => 0, 'unknown_columns' => $unknown,
            'issues' => [], 'preview' => [], 'rejected' => [], 'warnings' => $warnings,
            'header_line' => $headerLine,
        ];

        $this->categories = [];
        foreach (Category::query()->get() as $c) {
            $k1 = $this->norm((string) $c->name);
            $k2 = $this->norm((string) $c->slug);
            if ($k1 !== '') {
                $this->categories[$k1] = $c->id;
            }
            if ($k2 !== '') {
                $this->categories[$k2] = $c->id;
            }
        }

        $seen = [];
        DB::beginTransaction();
        try {
            foreach ($rows as [$line, $cells]) {
                $data = [];
                foreach ($map as $idx => $key) {
                    $data[$key] = $cells[$idx] ?? null;
                }
                $this->report['total']++;
                try {
                    $result = DB::transaction(fn () => $this->importRow($line, $data, $publishDefault, $admin, $seen));
                } catch (Throwable $e) {
                    report($e);
                    $this->issue($line, 'error', null, 'Erreur inattendue, ligne ignoree.', $data);
                    $result = 'skipped';
                }
                if ($result === 'skipped' && count($this->report['rejected']) < 500) {
                    $reason = '';
                    foreach (array_reverse($this->report['issues']) as $iss) {
                        if ($iss['row'] === $line && $iss['level'] === 'error') {
                            $reason = $iss['message'];
                            break;
                        }
                    }
                    $this->report['rejected'][] = ['row' => $line, 'reason' => $reason, 'data' => array_map(fn ($v) => $this->str($v), $data)];
                }
                $this->report[$result]++;
            }
        } finally {
            $dryRun ? DB::rollBack() : DB::commit();
        }

        if (! $dryRun && $this->opts['remember']) {
            $this->rememberMapping($headers, $map);
        }

        return $this->report;
    }

    private function importRow(int $line, array $d, bool $publishDefault, ?User $admin, array &$seen): string
    {
        $name = $this->str($d['name'] ?? null);
        $slugKey = $this->str($d['slug'] ?? null);
        $slugKey = $slugKey !== '' ? Str::slug($slugKey) : '';

        if ($name === '' && $slugKey === '') {
            $this->issue($line, 'error', 'name', 'Nom du produit manquant.', $d);

            return 'skipped';
        }

        $product = null;
        if ($slugKey !== '') {
            $product = Product::query()->without('images')->where('slug', $slugKey)->first();
        }
        if ($product === null && $name !== '') {
            $product = Product::query()->without('images')->whereRaw('LOWER(name) = ?', [mb_strtolower($name)])->first();
        }
        $creating = $product === null;

        if ($creating && $this->opts['mode'] === 'update') {
            $this->issue($line, 'error', null, 'Produit introuvable : ligne ignoree (mode mise a jour seulement).', $d);

            return 'skipped';
        }
        if (! $creating && $this->opts['mode'] === 'create') {
            $this->issue($line, 'warning', null, 'Produit deja existant : ligne ignoree (mode creation seulement).', $d);

            return 'unchanged';
        }

        if ($creating && $name === '') {
            $this->issue($line, 'error', 'name', 'Aucun produit avec ce slug et aucun nom fourni.', $d);

            return 'skipped';
        }
        if (mb_strlen($name) > 255) {
            $name = mb_substr($name, 0, 255);
            $this->issue($line, 'warning', 'name', 'Nom tronque a 255 caracteres.', $d);
        }
        if (! $creating && isset($seen[$product->id])) {
            $this->issue($line, 'warning', 'name', 'Doublon dans le fichier (deja vu ligne '.$seen[$product->id].') : cette ligne s applique par-dessus.', $d);
        }

        $attrs = [];

        // Prix
        if (! $this->blank($d['price'] ?? null)) {
            $p = $this->parseInt($d['price']);
            if ($p === null || $p < ($creating ? 1 : 0) || $p > 2000000000) {
                if ($creating) {
                    $this->issue($line, 'error', 'price', 'Prix invalide ("'.Str::limit($this->str($d['price']), 30).'").', $d);

                    return 'skipped';
                }
                $this->issue($line, 'warning', 'price', 'Prix invalide, ancien prix conserve.', $d);
            } else {
                $attrs['price'] = $p;
            }
        } elseif ($creating) {
            $this->issue($line, 'error', 'price', 'Prix manquant.', $d);

            return 'skipped';
        }

        // Stock
        $stock = null;
        if (! $this->blank($d['stock'] ?? null)) {
            $s = $this->parseInt($d['stock'], true);
            if ($s === null || $s < 0 || $s > 10000000) {
                $this->issue($line, 'warning', 'stock', 'Stock invalide ("'.Str::limit($this->str($d['stock']), 30).'"), ignore.', $d);
            } else {
                $stock = $s;
            }
        }

        // Champs texte
        foreach (['name_en' => 255, 'description' => 65000, 'description_en' => 65000] as $f => $max) {
            $v = $this->str($d[$f] ?? null);
            if ($v !== '') {
                $attrs[$f] = mb_substr($v, 0, $max);
            }
        }

        // Categorie (comparaison sans casse ni accents)
        $cat = $this->str($d['category'] ?? null);
        if ($cat !== '') {
            $key = $this->norm($cat);
            $id = $key !== '' ? ($this->categories[$key] ?? null) : null;
            if ($id === null) {
                if ($this->opts['create_categories'] && $key !== '') {
                    $catName = Str::limit($cat, 100, '');
                    $catSlug = Str::slug($catName) ?: 'categorie';
                    if (Category::query()->where('slug', $catSlug)->exists()) {
                        $catSlug .= '-'.Str::lower(Str::random(4));
                    }
                    $newCat = Category::query()->create(['name' => $catName, 'slug' => $catSlug, 'position' => 0, 'is_active' => true]);
                    $this->categories[$key] = $newCat->id;
                    $attrs['category_id'] = $newCat->id;
                    $this->issue($line, 'warning', 'category', 'Categorie creee : "'.$catName.'".', $d);
                } else {
                    $this->issue($line, 'warning', 'category', 'Categorie inconnue ("'.Str::limit($cat, 40).'"), ignoree.', $d);
                }
            } else {
                $attrs['category_id'] = $id;
            }
        }

        // Publication
        if (! $this->blank($d['published'] ?? null)) {
            $b = $this->parseBool($this->str($d['published']));
            if ($b === null) {
                $this->issue($line, 'warning', 'published', 'Valeur de publication illisible (attendu oui/non), ignoree.', $d);
            } else {
                $attrs['is_published'] = $b;
            }
        }

        // Lien image
        $img = $this->str($d['image_link'] ?? null);
        if ($img !== '') {
            if (filter_var($img, FILTER_VALIDATE_URL) && preg_match('#^https?://#i', $img) && mb_strlen($img) <= 2048) {
                $attrs['image_link'] = $img;
            } else {
                $this->issue($line, 'warning', 'image_link', 'Lien image invalide (http/https attendu), ignore.', $d);
            }
        }

        $stockService = app(StockService::class);

        if ($creating) {
            $slug = $slugKey !== '' ? $slugKey : Str::slug($name);
            if ($slug === '') {
                $slug = 'produit';
            }
            if (Product::query()->without('images')->where('slug', $slug)->exists()) {
                $slug .= '-'.Str::lower(Str::random(5));
            }
            $product = Product::query()->create(array_merge(
                ['name' => $name, 'slug' => $slug, 'stock' => 0, 'is_published' => $publishDefault],
                $attrs,
            ));
            if (($stock ?? 0) > 0) {
                $stockService->adjust($product, $stock, StockMovementReason::Initial, $admin, 'Import Excel');
            }
            $seen[$product->id] = $line;
            $this->preview($line, 'create', $name, $attrs['price'] ?? null, $stock ?? 0);

            return 'created';
        }

        if ($name !== '' && $slugKey !== '') {
            $attrs['name'] = $name;
        }
        $product->fill($attrs);
        $changed = $product->isDirty();
        if ($changed) {
            $product->save();
        }
        $stockChanged = false;
        if (! $this->opts['keep_stock'] && $stock !== null && $stock !== (int) $product->stock) {
            $stockService->setTo($product, $stock, StockMovementReason::Inventory, $admin, 'Import Excel');
            $stockChanged = true;
        }
        $seen[$product->id] = $line;

        $action = ($changed || $stockChanged) ? 'update' : 'unchanged';
        $this->preview($line, $action, $product->name, $attrs['price'] ?? (int) $product->price, $stock ?? (int) $product->stock);

        return $action === 'update' ? 'updated' : 'unchanged';
    }

class ProductImportController extends Controller

    private function extensionOf(Request $request): string
    {
        $ext = strtolower((string) $request->file('file')->getClientOriginalExtension());
        if (! in_array($ext, ['xlsx', 'csv', 'txt'], true)) {
            throw ValidationException::withMessages(['file' => 'Format non supporte : utilisez .xlsx ou .csv (pas .xls).']);
        }

        return $ext;
    }

    public function analyze(Request $request, ProductImporter $importer): JsonResponse
    {
        $this->authorize('create', Product::class);

        $request->validate([
            'file' => ['required', 'file', 'max:5120'],
            'sheet' => ['nullable', 'integer', 'min:0', 'max:200'],
            'header_line' => ['nullable', 'integer', 'min:1', 'max:500'],
        ]);
        $ext = $this->extensionOf($request);

        @set_time_limit(120);

        try {
            $data = $importer->analyze(
                $request->file('file')->getRealPath(),
                $ext,
                $request->filled('sheet') ? (int) $request->input('sheet') : null,
                $request->filled('header_line') ? (int) $request->input('header_line') : null,
            );
        } catch (\RuntimeException $e) {
            throw ValidationException::withMessages(['file' => $e->getMessage()]);
        }

        return response()->json(['data' => $data]);
    }

    public function import(Request $request, ProductImporter $importer): JsonResponse
    {
        $this->authorize('create', Product::class);

        $request->validate([
            'file' => ['required', 'file', 'max:5120'],
            'dry_run' => ['nullable', 'boolean'],
            'publish' => ['nullable', 'boolean'],
            'mode' => ['nullable', 'in:both,create,update'],
            'keep_stock' => ['nullable', 'boolean'],
            'create_categories' => ['nullable', 'boolean'],
            'mapping' => ['nullable', 'string', 'max:5000'],
            'sheet' => ['nullable', 'integer', 'min:0', 'max:200'],
            'header_line' => ['nullable', 'integer', 'min:1', 'max:500'],
            'remember' => ['nullable', 'boolean'],
        ]);
        $ext = $this->extensionOf($request);

        $mapping = null;
        if ($request->filled('mapping')) {
            $decoded = json_decode((string) $request->input('mapping'), true);
            if (! is_array($decoded)) {
                throw ValidationException::withMessages(['file' => 'Correspondance des colonnes invalide.']);
            }
            $mapping = $decoded;
        }

        @set_time_limit(180);

        try {
            $report = $importer->run(
                $request->file('file')->getRealPath(),
                $ext,
                $request->boolean('dry_run'),
                $request->boolean('publish'),
                $request->user(),
                [
                    'mode' => (string) $request->input('mode', 'both'),
                    'keep_stock' => $request->boolean('keep_stock'),
                    'create_categories' => $request->boolean('create_categories'),
                    'mapping' => $mapping,
                    'sheet' => $request->filled('sheet') ? (int) $request->input('sheet') : null,
                    'header_line' => $request->filled('header_line') ? (int) $request->input('header_line') : null,
                    'remember' => $request->boolean('remember'),
                ],
            );
        } catch (\RuntimeException $e) {
            throw ValidationException::withMessages(['file' => $e->getMessage()]);
        }

        return response()->json(['data' => $report]);
    }

    /** Unités détenues par des commandes en attente, par produit. */
    private function reservedByProduct(): \Illuminate\Database\Query\Builder
    {
        $types = array_unique([Product::class, (new Product)->getMorphClass()]);

        return DB::table('order_items')
            ->join('orders', 'orders.id', '=', 'order_items.order_id')
            ->where('orders.stock_reserved', true)
            ->where('orders.status', 'pending')
            ->whereIn('order_items.itemable_type', $types)
            ->groupBy('order_items.itemable_id')
            ->selectRaw('order_items.itemable_id as product_id, SUM(order_items.quantity) as qty');
    }

    private function filtered(Request $request, int $threshold): Builder
    {
        $q = trim((string) $request->query('q', ''));
        $filter = (string) $request->query('filter', '');
        $sort = (string) $request->query('sort', 'stock_asc');

        $query = Product::query()
            ->without('images')
            ->leftJoinSub($this->reservedByProduct(), 'r', 'r.product_id', '=', 'products.id')
            ->select('products.*')
            ->selectRaw('COALESCE(r.qty, 0) as reserved_qty');

        $query
            ->when($q !== '', function (Builder $w) use ($q): void {
                $like = '%'.mb_strtolower(addcslashes($q, '%_\\')).'%';
                $w->where(fn (Builder $x) => $x
                    ->whereRaw('LOWER(products.name) LIKE ?', [$like])
                    ->orWhereRaw("LOWER(COALESCE(products.name_en, '')) LIKE ?", [$like])
                    ->orWhereRaw('LOWER(products.slug) LIKE ?', [$like]));
            })
            ->when($filter === 'out', fn (Builder $w) => $w->where('products.stock', '<=', 0))
            ->when($filter === 'low', fn (Builder $w) => $w->where('products.stock', '>', 0)->where('products.stock', '<=', $threshold))
            ->when($filter === 'reserved', fn (Builder $w) => $w->where('r.qty', '>', 0))
            ->when($filter === 'unpublished', fn (Builder $w) => $w->where('products.is_published', false));

        match ($sort) {
            'stock_desc' => $query->orderByDesc('products.stock'),
            'name' => $query->orderBy('products.name'),
            'recent' => $query->orderByDesc('products.id'),
            default => $query->orderBy('products.stock'),
        };

        return $query->orderBy('products.id');
    }

    /** @return array<string, mixed> */
    private function row(Product $p, int $threshold): array
    {
        $stock = (int) $p->stock;

        return [
            'id' => $p->id,
            'name' => $p->name,
            'slug' => $p->slug,
            'image_url' => $p->imageUrl(),
            'price' => (int) $p->price,
            'stock' => $stock,
            'reserved' => (int) $p->reserved_qty,
            'value' => $stock * (int) $p->price,
            'status' => $stock <= 0 ? 'out' : ($stock <= $threshold ? 'low' : 'ok'),
            'is_published' => (bool) $p->is_published,
        ];
    }

    /** @return array<string, int> */
    private function summary(int $threshold): array
    {
        $a = DB::table('products')->selectRaw(
            'COUNT(*) as products_total,
            COALESCE(SUM(stock), 0) as units,
            COALESCE(SUM(stock * price), 0) as value,
            SUM(CASE WHEN stock <= 0 THEN 1 ELSE 0 END) as out_count,
            SUM(CASE WHEN stock > 0 AND stock <= ? THEN 1 ELSE 0 END) as low_count,
            SUM(CASE WHEN is_published AND stock <= 0 THEN 1 ELSE 0 END) as published_out',
            [$threshold],
        )->first();

        return [
            'products_total' => (int) ($a->products_total ?? 0),
            'units' => (int) ($a->units ?? 0),
            'value' => (int) ($a->value ?? 0),
            'out_count' => (int) ($a->out_count ?? 0),
            'low_count' => (int) ($a->low_count ?? 0),
            'published_out' => (int) ($a->published_out ?? 0),
            'reserved_units' => (int) DB::query()->fromSub($this->reservedByProduct(), 'r')->sum('qty'),
            'conflicts' => (int) DB::table('orders')->where('stock_conflict', true)->count(),
            'low_stock_threshold' => $threshold,
        ];
    }
}

import axios from "axios"
import i18n from "../i18n"
import type { ApiErrorBody } from "../types"

export type ErrorKind =
  | "offline" | "timeout" | "network" | "unauthorized" | "forbidden" | "not_found"
  | "expired" | "validation" | "rate_limit" | "server" | "unknown"

export interface ErrorInfo {
  kind: ErrorKind
  status?: number
  message: string
  retryable: boolean
}

const tr = (key: string, fallback: string) => i18n.t(key, { defaultValue: fallback })

export function getErrorInfo(error: unknown): ErrorInfo {
  if (!axios.isAxiosError<ApiErrorBody>(error)) {
    return { kind: "unknown", message: tr("errors.unexpected", "Une erreur inattendue est survenue."), retryable: false }
  }

  const res = error.response
  if (!res) {
    if (!navigator.onLine) return { kind: "offline", message: tr("errors.offline", "Vous êtes hors ligne. Vérifiez votre connexion."), retryable: true }
    if (error.code === "ECONNABORTED" || error.code === "ETIMEDOUT") return { kind: "timeout", message: tr("errors.timeout", "Le serveur met trop de temps à répondre."), retryable: true }
    return { kind: "network", message: tr("errors.network", "Impossible de joindre le serveur."), retryable: true }
  }

  const { status, data } = res
  const firstField = data?.errors ? Object.values(data.errors)[0]?.[0] : undefined
  const serverMessage = firstField ?? data?.message

  if (status === 401) return { kind: "unauthorized", status, message: tr("errors.session_expired", "Votre session a expiré. Reconnectez-vous."), retryable: false }
  if (status === 403) return { kind: "forbidden", status, message: serverMessage ?? tr("errors.forbidden", "Vous n'avez pas accès à cette ressource."), retryable: false }
  if (status === 404) return { kind: "not_found", status, message: tr("errors.not_found", "Cet élément est introuvable."), retryable: false }
  if (status === 419) return { kind: "expired", status, message: tr("errors.csrf", "La page a expiré. Rechargez-la puis réessayez."), retryable: false }
  if (status === 422) return { kind: "validation", status, message: serverMessage ?? tr("errors.validation", "Certaines informations sont invalides."), retryable: false }
  if (status === 429) return { kind: "rate_limit", status, message: tr("errors.rate_limit", "Trop de tentatives. Patientez un instant."), retryable: true }
  if (status >= 500) return { kind: "server", status, message: tr("errors.server", "Un problème est survenu de notre côté. Réessayez."), retryable: true }
  return { kind: "unknown", status, message: serverMessage ?? tr("errors.unexpected", "Une erreur inattendue est survenue."), retryable: false }
}

export const getErrorMessage = (error: unknown) => getErrorInfo(error).message

  const after = !valid ? null : reason === "restock" ? row.stock + n : reason === "damage" ? row.stock - n : reason === "inventory" ? n : row.stock + n

    mutationFn: () => adminApi.adjustStock(row.slug, { reason, quantity: n, note: note.trim() || undefined }),

  }

function QuickStock({ row, onDone, onError }: { row: StockRow; onDone: () => Promise<void>; onError: (m: string) => void }) {
  const [editing, setEditing] = useState(false)
  const [val, setVal] = useState(String(row.stock))

  const save = useMutation({
    mutationFn: (n: number) => adminApi.adjustStock(row.slug, { reason: "inventory", quantity: n, note: "Edition rapide" }),
    onSuccess: async () => { await onDone(); setEditing(false) },
    onError: (e) => onError(errMsg(e)),
  })

  function commit() {
    const n = Number(val)
    if (val.trim() === "" || !Number.isInteger(n) || n < 0) return onError("Entre un stock entier (0 ou plus).")
    if (n === row.stock) return setEditing(false)
    save.mutate(n)
  }

  if (!editing) {
    return (
      <button type="button" title="Cliquer pour modifier le stock" onClick={() => { setVal(String(row.stock)); setEditing(true) }}
        className="min-h-[36px] min-w-[3rem] rounded-md border border-transparent px-2 py-1 text-base font-bold hover:border-line hover:bg-page">
        {row.stock}
      </button>
    )
  }

  return (
    <div className="flex flex-wrap items-center justify-center gap-1">
      <input type="number" min={0} step={1} inputMode="numeric" autoFocus value={val} onChange={(e) => setVal(e.target.value)}
        onKeyDown={(e) => { if (e.key === "Enter") commit(); if (e.key === "Escape") setEditing(false) }}
        aria-label="Nouveau stock" className="min-h-[40px] w-20 rounded-lg border border-line bg-surface px-2 py-1 text-right outline-none focus:border-primary focus:ring-2 focus:ring-primary/20" />
      <button type="button" disabled={save.isPending} onClick={commit}
        className="min-h-[40px] rounded-lg bg-primary px-3 text-sm font-semibold text-white disabled:opacity-50">OK</button>
      <button type="button" onClick={() => setEditing(false)} aria-label="Annuler"
        className="min-h-[40px] rounded-lg border border-line bg-surface px-3 text-sm text-muted hover:text-ink">✕</button>
    </div>
  )
}

function errText(e: unknown): string {
  if (!isAxiosError(e)) return T.generic
  if (!e.response) return T.network
  if (e.response.status === 413) return T.tooBig
  const d = e.response.data as { message?: string; errors?: Record<string, string[]> } | undefined
  const first = d?.errors ? Object.values(d.errors)[0]?.[0] : undefined
  return first ?? d?.message ?? T.generic
}

function csvCell(v: unknown): string {
  let s = String(v ?? "")
  if (s !== "" && ["=", "+", "-", "@"].includes(s[0])) s = "'" + s
  return `"${s.replace(/"/g, '""')}"`
}

}
