    public function index(Request $request): JsonResponse
    {
        $this->authorize('viewAny', Course::class);

        $perPage = min(max($request->integer('per_page', 20), 1), 100);
        $q = trim((string) $request->query('q', ''));
        $status = (string) $request->query('status', '');

        $courses = Course::query()
            ->with('instructor:id,name')
            ->withCount('enrollments')
            ->withCount(['enrollments as completed_count' => fn ($w) => $w->where(
                fn ($x) => $x->where('progress', '>=', 100)->orWhereNotNull('completed_at'),
            )])
            ->withAvg('enrollments as avg_progress', 'progress')
            ->managedBy($request->user())
            ->when($status === 'published', fn ($w) => $w->where('is_published', true))
            ->when($status === 'draft', fn ($w) => $w->where('is_published', false))
            ->when($q !== '', fn ($w) => $w->where('title', 'like', '%'.addcslashes($q, '%_\\').'%'))
            ->latest('id')
            ->paginate($perPage);

        return response()->json([
            'data' => CourseResource::collection($courses),
            'meta' => [
                'current_page' => $courses->currentPage(),
                'last_page' => $courses->lastPage(),
                'total' => $courses->total(),
            ],
        ]);
    }

    public function store(Request $request): JsonResponse
    {
        $this->authorize('create', Course::class);

        $validated = $request->validate([
            'title' => ['required', 'string', 'max:255'],
            'slug' => ['sometimes', 'string', 'max:255', 'unique:courses,slug'],
            'description' => ['nullable', 'string'],
            'title_en' => ['nullable', 'string', 'max:255'],
            'description_en' => ['nullable', 'string'],
            'objectives' => ['nullable', 'string', 'max:10000'],
            'objectives_en' => ['nullable', 'string', 'max:10000'],
            'program' => ['nullable', 'string', 'max:20000'],
            'program_en' => ['nullable', 'string', 'max:20000'],
            'audience' => ['nullable', 'string', 'max:10000'],
            'audience_en' => ['nullable', 'string', 'max:10000'],
            'prerequisites' => ['nullable', 'string', 'max:10000'],
            'prerequisites_en' => ['nullable', 'string', 'max:10000'],
            'price' => ['required', 'integer', 'min:0'],
            'stream_video_id' => ['nullable', 'string', 'max:255'],
            'livekit_room' => ['nullable', 'string', 'max:255'],
            'duration_seconds' => ['nullable', 'integer', 'min:1', 'max:86400'],
            'is_published' => ['sometimes', 'boolean'],
            'instructor_id' => ['sometimes', 'nullable', Rule::exists('users', 'id')->where('is_admin', true)->where('admin_scope', 'instructor')],

            'product_id' => ['sometimes', 'nullable', 'integer', 'exists:products,id'],
        ]);

        if (! isset($validated['slug'])) {
            $validated['slug'] = Str::slug($validated['title']).'-'.Str::random(6);
        }

        // Un formateur est toujours propriétaire de ce qu'il crée
        if ($request->user()->isInstructorOnly()) {
            $validated['instructor_id'] = $request->user()->id;
        }

        // pl-store
        $hasProduct = array_key_exists('product_id', $validated);
        $productId = $validated['product_id'] ?? null;
        unset($validated['product_id']);

        $course = Course::query()->create($validated);

        if ($hasProduct && ! $request->user()->isInstructorOnly()) {
            $this->linkProduct($course, $productId);
        }

        return response()->json([
            'data' => CourseResource::make($course),
        ], 201);
    }

    /** Un seul produit par cours : on delie l'ancien puis on lie le nouveau. */
    private function linkProduct(Course $course, ?int $productId): void
    {
        \Illuminate\Support\Facades\DB::transaction(function () use ($course, $productId): void {
            \App\Models\Product::query()->where('course_id', $course->id)->update(['course_id' => null]);
            if ($productId !== null) {
                \App\Models\Product::query()->whereKey($productId)->update(['course_id' => $course->id]);
            }
        });
    }

    public function index(Request $request): JsonResponse
    {
        $this->authorize('viewAny', Product::class);

        $filter = (string) $request->query('stock', '');
        $threshold = (int) config('shop.low_stock_threshold', 5);

        $products = Product::query()
            ->when($filter === 'out', fn ($q) => $q->where('stock', '<=', 0))
            ->when($filter === 'low', fn ($q) => $q->where('stock', '>', 0)->where('stock', '<=', $threshold))
            ->when(trim((string) $request->query('q', '')) !== '', function ($q) use ($request): void {
                $like = '%'.mb_strtolower(addcslashes(trim((string) $request->query('q')), '%_\\')).'%';
                $q->where(fn ($x) => $x
                    ->whereRaw('LOWER(name) LIKE ?', [$like])
                    ->orWhereRaw("LOWER(COALESCE(name_en, '')) LIKE ?", [$like])
                    ->orWhereRaw('LOWER(slug) LIKE ?', [$like]));
            })
            ->when(trim((string) $request->query('category', '')) !== '', function ($q) use ($request): void {
                $c = trim((string) $request->query('category'));
                $c === 'none' ? $q->whereNull('category_id') : $q->where('category_id', (int) $c);
            })
            ->when($request->query('published') === '1', fn ($q) => $q->where('is_published', true))
            ->when($request->query('published') === '0', fn ($q) => $q->where('is_published', false))
            ->when($request->query('sort') === 'stock', fn ($q) => $q->orderBy('stock'))
            ->latest('id')
            ->paginate(20);

        return response()->json([
            'data' => ProductResource::collection($products),
            'meta' => [
                'current_page' => $products->currentPage(),
                'last_page' => $products->lastPage(),
                'total' => $products->total(),
                'low_stock_threshold' => $threshold,
                'low_stock_count' => Product::query()->without('images')->where('stock', '>', 0)->where('stock', '<=', $threshold)->count(),

namespace App\Support;

use Illuminate\Http\UploadedFile;
use Illuminate\Support\Facades\Storage;
use Illuminate\Support\Str;
use Throwable;

/**
 * Redimensionne et convertit en WebP (image 1600 px + miniature 480 px).
 * Fichiers produits : <id>_p.webp et <id>_t.webp. Sans GD/WebP, le fichier est stocke tel quel.
 */
final class ImageProcessor
{
    private const MAIN = 1600;
    private const THUMB = 480;
    private const QUALITY = 82;
    private const MAX_PIXELS = 40000000;

    public static function store(UploadedFile $file, string $dir, ?string $disk = null): string
    {
        $disk ??= MediaDisk::name();
        $dir = trim($dir, '/');

        if (self::available()) {
            try {
                $out = self::process((string) $file->getRealPath());
                if ($out !== null) {
                    $id = Str::random(32);
                    $fs = Storage::disk($disk);
                    $main = $dir.'/'.$id.'_p.webp';
                    $fs->put($main, $out[0]);
                    $fs->put($dir.'/'.$id.'_t.webp', $out[1]);

                    return $main;
                }
            } catch (Throwable $e) {
                report($e);
            }
        }

        return (string) $file->store($dir, ['disk' => $disk]);
    }

    /** Supprime le fichier et sa miniature eventuelle. */
    public static function delete(?string $path, ?string $disk = null): void
    {
        if (! is_string($path) || $path === '') {
            return;
        }
        try {
            $fs = Storage::disk($disk ?: 'public');
            $fs->delete($path);
            if (str_ends_with($path, '_p.webp')) {
                $fs->delete(substr($path, 0, -7).'_t.webp');
            }
        } catch (Throwable $e) {
            report($e);
        }
    }

    /** Chemin de la miniature, ou null si le fichier n'en a pas. */
    public static function thumbPath(string $path): ?string
    {
        return str_ends_with($path, '_p.webp') ? substr($path, 0, -7).'_t.webp' : null;
    }

    private static function available(): bool
    {
        return extension_loaded('gd') && function_exists('imagewebp');
    }

    /** @return array{0: string, 1: string}|null */
    private static function process(string $file): ?array
    {
        $info = @getimagesize($file);
        if (! $info || ($info[0] * $info[1]) > self::MAX_PIXELS) {
            return null;
        }

        $img = match ($info['mime'] ?? '') {
            'image/jpeg' => @imagecreatefromjpeg($file),
            'image/png' => @imagecreatefrompng($file),
            'image/webp' => @imagecreatefromwebp($file),
            default => false,
        };
        if (! $img) {
            return null;
        }

        if (($info['mime'] ?? '') === 'image/jpeg' && function_exists('exif_read_data')) {
            $exif = @exif_read_data($file);
            $angle = match ((int) ($exif['Orientation'] ?? 1)) {
                3 => 180,
                6 => -90,
                8 => 90,
                default => 0,
            };
            if ($angle !== 0) {
                $rot = imagerotate($img, $angle, 0);
                if ($rot) {
                    $img = $rot;
                }
            }
        }

        if (! imageistruecolor($img)) {
            imagepalettetotruecolor($img);
        }
        imagealphablending($img, false);
        imagesavealpha($img, true);

        $main = self::fit($img, self::MAIN);
        $thumb = self::fit($main, self::THUMB);

        return [self::encode($main), self::encode($thumb)];
    }
