<?php

namespace App\Http\Resources;

use App\Models\Product;
use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\JsonResource;

/** @mixin Product */
class ProductResource extends JsonResource
{
    /**
     * @return array<string, mixed>
     */
    public function toArray(Request $request): array
    {
        // L'API admin renvoie les valeurs brutes (a editer) ; la boutique recoit la langue courante.
        $admin = $request->is('api/v1/admin/*');

        return [
            'id' => $this->id,
            'name' => $admin ? $this->name : $this->localizedName(),
            'slug' => $this->slug,
            'description' => $admin ? $this->description : $this->localizedDescription(),
            'name_en' => $this->when($admin, $this->name_en),
            'description_en' => $this->when($admin, $this->description_en),
            'image_url' => $this->imageUrl(),
            'images' => array_values(array_filter([
                $this->imageUrl(),
                ...$this->images->map(fn ($image) => $image->url())->all(),
            ])),
            'thumbs' => array_values(array_filter([
                $this->thumbUrl(),
                ...$this->images->map(fn ($image) => $image->thumbUrl())->all(),
            ])),
            'specs' => $admin ? $this->specs : $this->localizedExtra('specs'),
            'delivery_info' => $admin ? $this->delivery_info : $this->localizedExtra('delivery_info'),
            'specs_en' => $this->when($admin, $this->specs_en),
            'delivery_info_en' => $this->when($admin, $this->delivery_info_en),
            'share_url' => rtrim(config('app.url'), '/').'/p/'.$this->slug,
            'image_link' => $this->when($admin, $this->image_link),
            'gallery' => $this->when($admin, fn () => $this->images->map(fn ($image) => ['id' => $image->id, 'url' => $image->url()])->values()->all()),
            'category_id' => $this->category_id,
            'course_id' => $this->when($admin, $this->course_id),
            'category' => $this->category ? [
                'id' => $this->category->id,
                'name' => $admin ? $this->category->name : $this->category->localizedName(),
                'slug' => $this->category->slug,
            ] : null,
            'price' => $this->price,
            'stock' => $this->when($request->user()?->is_admin, $this->stock),
            'in_stock' => (int) $this->stock > 0,
            'low_stock' => ((int) $this->stock > 0 && (int) $this->stock <= (int) config('shop.low_stock_threshold', 5)) ? (int) $this->stock : null,
            'max_quantity' => max(0, min(99, (int) $this->stock)),
            'is_published' => $this->when($request->user()?->is_admin, $this->is_published),
            'created_at' => $this->created_at,
            'updated_at' => $this->updated_at,
        ];
    }
}

/** Champs de chaque etape (sert a la validation et au saut vers l'etape en erreur). */
const STEP_FIELDS: string[][] = [
  ["course_id", "name"],
  ["gender", "birth_date", "birth_country", "birth_place", "nationality"],
  ["phone", "country", "city", "address", "profession"],

const LAST = STEP_FIELDS.length - 1

function Stepper({ step, labels }: { step: number; labels: string[] }) {
  const pct = (step / (labels.length - 1)) * 100
  return (
    <nav aria-label="Progression" className="mb-1">
      <div className="relative mx-4">
        <div className="absolute left-0 right-0 top-4 h-1 -translate-y-1/2 rounded-full bg-line" />
        <div
          className="absolute left-0 top-4 h-1 -translate-y-1/2 rounded-full bg-primary transition-all duration-500"
          style={{ width: `${pct}%` }}
        />
        <ol className="relative flex justify-between">
          {labels.map((l, i) => {
            const done = i < step
            const current = i === step
            return (
              <li key={l} className="flex w-0 flex-col items-center" aria-current={current ? "step" : undefined}>
                <span
                  className={`flex h-8 w-8 shrink-0 items-center justify-center rounded-full border-2 text-sm font-bold transition-colors ${
                    done
                      ? "border-primary bg-primary text-white"
                      : current
                        ? "border-primary bg-surface text-primary ring-4 ring-primary/15"
                        : "border-line bg-surface text-muted"
                  }`}
                >
                  {done ? "✓" : i + 1}
                </span>
                <span className={`mt-1.5 hidden whitespace-nowrap text-xs font-semibold sm:block ${current ? "text-primary" : "text-muted"}`}>{l}</span>
              </li>
            )
          })}
        </ol>
      </div>
      <p className="mt-3 text-center text-sm font-semibold text-primary sm:hidden">
        {step + 1} / {labels.length} · {labels[step]}
      </p>
    </nav>
  )
}

            : t("px.signup.create", { defaultValue: "Créer un compte" })}
        </button>
      </p>
    </div>,
  )
}

import { useEffect, useId, useMemo, useRef, useState, type KeyboardEvent } from "react"
import { useTranslation } from "react-i18next"
import type { CountryOption } from "../lib/geo"

interface Opt { key: string; value: string; label: string; sub?: string; divider?: boolean }

/** Compare sans accents ni majuscules. */
const norm = (s: string) => s.normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase().trim()

function Autocomplete(props: {
  label: string
  text: string
  onText: (v: string) => void
  options: Opt[]
  onPick: (o: Opt) => void
  onCommit?: () => void
  error?: string
  placeholder?: string
  busy?: boolean
  chevron?: boolean
  firstActive?: boolean
  emptyText?: string
}) {
  const { label, text, onText, options, onPick, onCommit, error, placeholder, busy, chevron, firstActive, emptyText } = props
  const uid = useId()
  const [open, setOpen] = useState(false)
  const [active, setActive] = useState(-1)
  const listRef = useRef<HTMLUListElement>(null)

  useEffect(() => {
    if (open && active >= 0) listRef.current?.children[active]?.scrollIntoView({ block: "nearest" })
  }, [active, open])

  function pick(o: Opt) {
    onPick(o)
    setOpen(false)
    setActive(-1)
  }

  function onKeyDown(e: KeyboardEvent<HTMLInputElement>) {
    if (e.key === "ArrowDown") {
      e.preventDefault()
      if (!open) setOpen(true)
      else setActive((i) => Math.min(options.length - 1, i + 1))
    } else if (e.key === "ArrowUp") {
      e.preventDefault()
      setActive((i) => Math.max(0, i - 1))
    } else if (e.key === "Enter") {
      if (open && active >= 0 && options[active]) {
        e.preventDefault()
        pick(options[active])
      }
    } else if (e.key === "Escape" && open) {
      e.preventDefault()
      e.stopPropagation()
      setOpen(false)
    }
  }

  const showList = open && (options.length > 0 || !!emptyText)

  return (
    <div>
      <label className="block text-sm font-medium" htmlFor={`${uid}-in`}>{label}</label>
      <div className="relative mt-1">
        <input
          id={`${uid}-in`}
          role="combobox"
          aria-expanded={showList}
          aria-controls={`${uid}-list`}
          aria-autocomplete="list"
          aria-activedescendant={active >= 0 ? `${uid}-o${active}` : undefined}
          aria-invalid={error ? true : undefined}
          aria-describedby={error ? `${uid}-err` : undefined}
          autoComplete="off"
          autoCapitalize="off"
          spellCheck={false}
          placeholder={placeholder}
          value={text}
          onChange={(e) => { onText(e.target.value); setOpen(true); setActive(firstActive ? 0 : -1) }}
          onFocus={() => setOpen(true)}
          onClick={() => setOpen(true)}
          onBlur={() => { setOpen(false); setActive(-1); onCommit?.() }}
          onKeyDown={onKeyDown}
          className={`min-h-[44px] w-full rounded-lg border bg-surface px-3 py-2 pr-9 text-sm outline-none transition focus:ring-2 ${
            error ? "border-danger focus:border-danger focus:ring-danger/20" : "border-line focus:border-primary focus:ring-primary/20"
          }`}
        />
        {busy ? (
          <span aria-hidden="true" className="absolute right-3 top-1/2 h-4 w-4 -translate-y-1/2 animate-spin rounded-full border-2 border-line border-t-primary" />
        ) : chevron ? (
          <span aria-hidden="true" className="pointer-events-none absolute right-3 top-1/2 -translate-y-1/2 text-xs text-muted">▾</span>
        ) : null}
        {showList && (
          <ul
            id={`${uid}-list`} role="listbox" ref={listRef}
            className="absolute z-30 mt-1 max-h-60 w-full overflow-auto rounded-xl border border-line bg-surface py-1 text-sm shadow-card-lg"
          >
            {options.length === 0 && <li className="px-3 py-2.5 text-muted">{emptyText}</li>}
            {options.map((o, i) => (
              <li
                key={o.key} id={`${uid}-o${i}`} role="option" aria-selected={i === active}
                onMouseDown={(e) => e.preventDefault()}
                onMouseMove={() => { if (active !== i) setActive(i) }}
                onClick={() => pick(o)}
                className={`flex cursor-pointer items-center justify-between gap-3 px-3 py-2.5 ${o.divider ? "border-t border-line" : ""} ${i === active ? "bg-primary/10 text-primary" : ""}`}
              >
                <span className="truncate">{o.label}</span>
                {o.sub && <span className="shrink-0 truncate text-xs text-muted">{o.sub}</span>}
              </li>
            ))}
          </ul>
        )}
      </div>
      {error && <p id={`${uid}-err`} className="mt-1 text-xs text-danger">{error}</p>}
    </div>
  )
}

export function CountrySelect(props: {
  label: string
  placeholder: string
  value: string
  onChange: (v: string) => void
  error?: string
  countries: { top: CountryOption[]; all: CountryOption[] }
}) {
  const { label, placeholder, value, onChange, error, countries } = props
  const { t } = useTranslation()
  const byCode = useMemo(() => new Map([...countries.top, ...countries.all].map((c) => [c.code, c.name])), [countries])
  const selectedName = byCode.get(value) ?? ""
  const [text, setText] = useState(selectedName)
  const [typing, setTyping] = useState(false)

  // Resynchronise quand la valeur change de l'exterieur (pre-remplissage, langue...).
  useEffect(() => { setText(selectedName); setTyping(false) }, [selectedName])

  const options = useMemo<Opt[]>(() => {
    const q = typing ? norm(text) : ""
    if (!q) {
      const topCodes = new Set(countries.top.map((c) => c.code))
      return [
        ...countries.top.map((c) => ({ key: `t-${c.code}`, value: c.code, label: c.name })),
        ...countries.all.filter((c) => !topCodes.has(c.code)).map((c, i) => ({ key: c.code, value: c.code, label: c.name, divider: i === 0 })),
      ]
    }
    const starts: Opt[] = []
    const contains: Opt[] = []
    for (const c of countries.all) {
      const n = norm(c.name)
      const o = { key: c.code, value: c.code, label: c.name }
      if (n.startsWith(q)) starts.push(o)
      else if (n.includes(q)) contains.push(o)
    }
    return [...starts, ...contains]
  }, [countries, text, typing])

  function commit() {
    if (!typing) return
    const q = norm(text)
    const exact = countries.all.find((c) => norm(c.name) === q)
    const only = options.length === 1 ? options[0] : undefined
    if (exact) { onChange(exact.code); setText(exact.name) }
    else if (only) { onChange(only.value); setText(only.label) }
    else setText(selectedName)
    setTyping(false)
  }

  return (
    <Autocomplete
      label={label} placeholder={placeholder} text={text} options={options} error={error} chevron firstActive
      emptyText={t("px.signup.no_result", { defaultValue: "Aucun résultat" })}
      onText={(v) => { setText(v); setTyping(true); if (!v.trim()) onChange("") }}
      onPick={(o) => { onChange(o.value); setText(o.label); setTyping(false) }}
      onCommit={commit}
    />
  )
}

const CACHE = new Map<string, Opt[]>()

/** Suggestions de villes via Open-Meteo (anti-rebond, annulation, cache). Echec silencieux. */
function useRemoteCities(query: string, country: string, lang: string, enabled: boolean) {
  const q = query.trim()
  const key = `${lang}|${country}|${norm(q)}`
  const [items, setItems] = useState<Opt[]>([])
  const [busy, setBusy] = useState(false)

  useEffect(() => {
    if (!enabled || q.length < 2) { setBusy(false); return }
    const hit = CACHE.get(key)
    if (hit) { setItems(hit); setBusy(false); return }
    const ctrl = new AbortController()
    setBusy(true)
    const timer = window.setTimeout(async () => {
      try {
        const qs = new URLSearchParams({ name: q, count: "8", language: lang, format: "json" })
        if (country) qs.set("countryCode", country)
        const res = await fetch(`https://geocoding-api.open-meteo.com/v1/search?${qs.toString()}`, { signal: ctrl.signal })
        if (!res.ok) throw new Error(String(res.status))
        const json = (await res.json()) as { results?: { id: number; name: string; admin1?: string; country?: string }[] }
        const list: Opt[] = (json.results ?? []).map((r) => ({
          key: `r-${r.id}`,
          value: r.name,
          label: r.name,
          sub: [r.admin1, country ? "" : r.country].filter(Boolean).join(", ") || undefined,
        }))
        CACHE.set(key, list)
        setItems(list)
      } catch {
        // Hors ligne, quota, abandon : on garde la liste locale et la saisie libre.
      } finally {
        if (!ctrl.signal.aborted) setBusy(false)
      }
    }, 300)
    return () => { window.clearTimeout(timer); ctrl.abort() }
  }, [enabled, q, key, country, lang])

  return { items: enabled && q.length >= 2 ? items : [], busy }
}

export function CityInput(props: {
  id?: string
  label: string
  value: string
  onChange: (v: string) => void
  error?: string
  cities: string[]
  country?: string
}) {
  const { label, value, onChange, error, cities, country } = props
  const { i18n } = useTranslation()
  const lang = (i18n.language || "fr").slice(0, 2)
  const [typing, setTyping] = useState(false)
  const remote = useRemoteCities(value, country ?? "", lang, typing)

  const options = useMemo<Opt[]>(() => {
    const q = norm(value)
    const starts: string[] = []
    const contains: string[] = []
    for (const c of cities) {
      const n = norm(c)
      if (!q || n.startsWith(q)) starts.push(c)
      else if (n.includes(q)) contains.push(c)
    }
    const local: Opt[] = [...starts, ...contains].slice(0, q ? 6 : 8).map((c) => ({ key: `l-${c}`, value: c, label: c }))
    const seen = new Set(local.map((o) => norm(o.label)))
    return [...local, ...remote.items.filter((o) => !seen.has(norm(o.label)))]
  }, [cities, value, remote.items])

  return (
    <Autocomplete
      label={label} text={value} options={options} error={error} busy={remote.busy}
      onText={(v) => { onChange(v); setTyping(true) }}
      onPick={(o) => { onChange(o.value); setTyping(false) }}
      onCommit={() => {
        setTyping(false)
        const clean = value.replace(/\s+/g, " ").trim()
        if (clean !== value) onChange(clean)
      }}
    />
  )
}

import { create } from "zustand"

export interface RecentProduct {
  id: number
  slug: string
  name: string
  price: number
  image: string | null
  in_stock?: boolean
}

const KEY = "recent_products"
const MAX = 8

function load(): RecentProduct[] {
  try {
    const raw = localStorage.getItem(KEY)
    const list: unknown = raw ? JSON.parse(raw) : []
    if (!Array.isArray(list)) return []
    return (list as RecentProduct[]).filter((x) => x && typeof x.slug === "string").slice(0, MAX)
  } catch {
    return []
  }
}

function save(items: RecentProduct[]) {
  try {
    localStorage.setItem(KEY, JSON.stringify(items))
  } catch {
    // stockage indisponible (navigation privee, quota) : on ignore
  }
}

interface RecentState {
  items: RecentProduct[]
  push: (p: RecentProduct) => void
  clear: () => void
}

// Memoire locale a l'appareil : rien n'est envoye au serveur
export const useRecentStore = create<RecentState>((set, get) => ({
  items: load(),
  push: (p) => {
    const items = [p, ...get().items.filter((x) => x.id !== p.id)].slice(0, MAX)
    save(items)
    set({ items })
  },
  clear: () => {
    save([])
    set({ items: [] })
  },
}))

      {q && (
        <p className="mb-3 flex flex-wrap items-center gap-2 text-sm text-muted" aria-live="polite">
          {list.isFetching ? "Recherche…" : `${list.data?.meta.total ?? 0} résultat(s) pour « ${q} » dans tous les onglets.`}
          <button type="button" onClick={() => setInput("")} className="font-semibold text-primary hover:underline">Effacer</button>
        </p>
      )}

      <div className="mb-4 grid gap-3 sm:grid-cols-2 xl:grid-cols-4">
        <div className="relative sm:col-span-2">
          <span className="pointer-events-none absolute inset-y-0 left-3 flex items-center text-muted"><Svg d={I_SEARCH} className="h-4 w-4" /></span>
          <input data-tour="pay-search" type="search" value={input} onChange={(e) => setInput(e.target.value)} placeholder="Transaction, référence, client, « payé », « cash », 15/10/2026 00:41…" className={`${inputClass} pl-9`} />
        </div>
        <label className={labelClass}>Du
          <input type="date" value={from} onChange={(e) => { setFrom(e.target.value); setPage(1) }} className={`${inputClass} mt-1`} />
        </label>
        <label className={labelClass}>Au
          <input type="date" value={to} onChange={(e) => { setTo(e.target.value); setPage(1) }} className={`${inputClass} mt-1`} />
        </label>
      </div>

      {list.isLoading && <p className="py-8 text-center text-muted">Chargement…</p>}
      {list.error && (
        <p role="alert" className="py-4 text-danger">{errMsg(list.error)} <button type="button" className="font-semibold underline" onClick={() => void list.refetch()}>Réessayer</button></p>
      )}
      {list.data && rows.length === 0 && <p className="rounded-card border border-dashed border-line py-10 text-center text-muted">Rien à traiter ici.</p>}

      {rows.length > 0 && (
        <>
          {/* Mobile et tablette : cartes */}
          <ul data-tour="pay-list" className="grid gap-3 md:grid-cols-2 xl:hidden">
            {rows.map((r) => (
              <li key={r.id} className="min-w-0 rounded-card border border-line bg-surface p-3.5 shadow-card">
                <div className="flex items-start justify-between gap-3">
                  <div className="min-w-0">
                    <p className="font-semibold">#{r.id}{r.public_ref ? ` · ${r.public_ref}` : ""}</p>
                    <p className="truncate text-sm text-muted">{r.customer ?? "Client invité"}{r.phone ? ` · ${r.phone}` : ""}</p>
                  </div>
                  <p className="money shrink-0 text-right text-base font-extrabold">{formatPrice(r.total, r.currency)}</p>
                </div>
                <div className="mt-2"><StatusPill r={r} /></div>
                <div className="mt-2"><PaymentInfo r={r} /></div>
                <p className="mt-2 text-xs text-muted">Commande du {fmtDate(r.created_at)}</p>
                <div className="mt-3 border-t border-line pt-3"><RowActions r={r} a={actions} /></div>
              </li>
            ))}
          </ul>

          {/* Grand écran : tableau */}
          <div data-tour="pay-list" className="hidden overflow-x-auto rounded-card border border-line bg-surface shadow-card xl:block">
            <table className="w-full min-w-[980px] text-left text-sm">
              <thead className="border-b border-line bg-page text-xs uppercase tracking-wide text-muted">
                <tr>
                  <th className="px-4 py-3 font-semibold">Commande</th>
                  <th className="px-4 py-3 font-semibold">Client</th>
                  <th className="px-4 py-3 text-right font-semibold">Montant</th>
                  <th className="px-4 py-3 font-semibold">Statut</th>
                  <th className="px-4 py-3 font-semibold">Paiement</th>
                  <th className="px-4 py-3 text-right font-semibold">Actions</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-line">
                {rows.map((r) => (
                  <tr key={r.id} className="align-top transition hover:bg-page/60">
                    <td className="whitespace-nowrap px-4 py-3">
                      <p className="font-semibold">#{r.id}</p>
                      <p className="text-xs text-muted">{fmtDate(r.created_at)}</p>
                    </td>
                    <td className="px-4 py-3">
                      <p className="max-w-[22ch] truncate font-medium">{r.customer ?? "Client invité"}</p>
                      <p className="text-xs text-muted">{r.phone ?? r.email ?? ""}</p>
                    </td>
                    <td className="whitespace-nowrap px-4 py-3 text-right font-bold">{formatPrice(r.total, r.currency)}</td>
                    <td className="px-4 py-3"><StatusPill r={r} /></td>
                    <td className="max-w-[34ch] px-4 py-3"><PaymentInfo r={r} /></td>
                    <td className="px-4 py-3"><div className="flex justify-end"><RowActions r={r} a={actions} /></div></td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </>
      )}

      {list.data && <Pager page={list.data.meta.current_page} last={list.data.meta.last_page} onChange={setPage} />}

      {markFor && markOrder && (
        <MarkPaidDialog key={markFor.id} order={markOrder} pending={markPaid.isPending} error={markError}
          onSubmit={(payload) => markPaid.mutate({ id: markFor.id, payload, hadAnomaly: Boolean(markFor.anomaly) })} onClose={() => setMarkFor(null)} />
      )}
      {reasonFor && (
        <ReasonDialog key={`${reasonFor.kind}-${reasonFor.row.id}`} kind={reasonFor.kind} row={reasonFor.row} pending={reasonAction.isPending} error={reasonError}
          onSubmit={(text) => { setReasonError(""); reasonAction.mutate({ kind: reasonFor.kind, row: reasonFor.row, text }) }} onClose={() => setReasonFor(null)} />
      )}
      {historyFor && <HistoryDrawer key={historyFor.id} row={historyFor} onClose={() => setHistoryFor(null)} />}
    </div>
  )
}
