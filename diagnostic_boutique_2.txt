<?php

use App\Http\Middleware\EnsureUserIsAdmin;
use App\Http\Middleware\OptionalSanctumAuth;
use Illuminate\Auth\AuthenticationException;
use Illuminate\Database\Eloquent\ModelNotFoundException;
use Illuminate\Foundation\Application;
use Illuminate\Foundation\Configuration\Exceptions;
use Illuminate\Foundation\Configuration\Middleware;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Log;
use Illuminate\Validation\ValidationException;
use Symfony\Component\HttpKernel\Exception\HttpExceptionInterface;
use Symfony\Component\HttpKernel\Exception\NotFoundHttpException;

return Application::configure(basePath: dirname(__DIR__))
    ->withRouting(
        web: __DIR__.'/../routes/web.php',
        api: __DIR__.'/../routes/api.php',
        commands: __DIR__.'/../routes/console.php',
        health: '/up',
    )
    ->withMiddleware(function (Middleware $middleware): void {
        $middleware->trustProxies(at: '*');
        $middleware->appendToGroup('api', \App\Http\Middleware\SetLocale::class);
        $middleware->alias([
            'admin' => EnsureUserIsAdmin::class,
            'admin.scope' => \App\Http\Middleware\EnsureAdminScope::class,
            'student' => \App\Http\Middleware\EnsureUserIsStudent::class,
            'not.blocked' => \App\Http\Middleware\EnsureUserIsNotBlocked::class,
            'sanctum.optional' => OptionalSanctumAuth::class,
        ]);
    })
    ->withExceptions(function (Exceptions $exceptions): void {
        // Toute exception non geree explicitement ailleurs est journalisee
        // avec son contexte, puis renvoyee au client sous une forme propre
        // et sans detail technique interne (jamais de stack trace exposee).
        $exceptions->render(function (Throwable $e, Request $request) {
            // Laravel sait deja produire un 401 / 422 JSON propre pour ces cas
            if ($e instanceof AuthenticationException
                || $e instanceof ValidationException) {
                return null;
            }
            if (! $request->expectsJson() && ! $request->is('api/*')) {
                return null; // laisse Laravel gerer les routes web normalement
            }

            $status = $e instanceof HttpExceptionInterface
                ? $e->getStatusCode()
                : 500;

            if ($status >= 500) {
                Log::error('Exception non geree', [
                    'message' => $e->getMessage(),
                    'exception' => $e::class,
                    'url' => $request->fullUrl(),
                    'method' => $request->method(),
                ]);
            }

            $message = $status >= 500
                ? 'Une erreur inattendue est survenue. Notre equipe a ete alertee.'
                : ($e instanceof NotFoundHttpException
                    && ($e->getMessage() === ''
                        || $e->getPrevious() instanceof ModelNotFoundException
                        || str_starts_with($e->getMessage(), 'The route '))
                    ? __('api.not_found')
                    : $e->getMessage());

            return response()->json(['message' => $message], $status);
        });
    })->create();

<?php

namespace App\Http\Controllers\Storefront;

use App\Http\Controllers\Controller;
use App\Http\Resources\CourseResource;
use App\Http\Resources\ProductResource;
use App\Models\Course;
use App\Models\Product;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class CatalogController extends Controller
{
    public function courses(Request $request): JsonResponse
    {
        $courses = Course::query()
            ->with('instructor:id,name')
            ->where('is_published', true)
            ->latest('id')
            ->paginate(min(max($request->integer('per_page', 20), 1), 100));

        return response()->json([
            'data' => CourseResource::collection($courses),
            'meta' => [
                'current_page' => $courses->currentPage(),
                'last_page' => $courses->lastPage(),
                'total' => $courses->total(),
            ],
        ]);
    }

    public function course(Request $request, Course $course): JsonResponse
    {
        $this->authorize('view', $course);

        return response()->json([
            'data' => CourseResource::make($course),
        ]);
    }

    public function products(Request $request): JsonResponse
    {
        $search = trim((string) $request->query('search', ''));
        $sort = (string) $request->query('sort', 'new');

        $query = Product::query()->where('is_published', true);

        if ($search !== '') {
            $like = '%'.str_replace(['%', '_'], ['\%', '\_'], $search).'%';
            $query->where(function ($q) use ($like) {
                $q->where('name', 'like', $like)->orWhere('name_en', 'like', $like);
            });
        }

        $cat = trim((string) $request->query('category', ''));
        if ($cat !== '') {
            $query->whereHas('category', fn ($q) => $q->where('slug', $cat)->where('is_active', true));
        }

        $minPrice = $request->query('min_price');
        $maxPrice = $request->query('max_price');
        if (is_numeric($minPrice) && (int) $minPrice > 0) {
            $query->where('price', '>=', (int) $minPrice);
        }
        if (is_numeric($maxPrice) && (int) $maxPrice > 0) {
            $query->where('price', '<=', (int) $maxPrice);
        }

        if ($request->boolean('in_stock')) {
            $query->where('stock', '>', 0);
        }
        if ($request->boolean('new')) {
            $query->where('created_at', '>=', now()->subDays(14));
        }

        match ($sort) {
            'price_asc' => $query->orderBy('price')->orderByDesc('id'),
            'price_desc' => $query->orderByDesc('price')->orderByDesc('id'),
            default => $query->latest('id'),
        };

        $products = $query->paginate(20)->withQueryString();

        return response()->json([
            'data' => ProductResource::collection($products),
            'meta' => [
                'current_page' => $products->currentPage(),
                'last_page' => $products->lastPage(),
                'total' => $products->total(),
            ],
        ]);
    }

    public function product(Request $request, Product $product): JsonResponse
    {
        $this->authorize('view', $product);

        return response()->json([
            'data' => ProductResource::make($product),
        ]);
    }
}

    }

<?php

namespace App\Http\Controllers\Api;

use App\Enums\OrderStatus;
use App\Exceptions\InsufficientStockException;
use App\Http\Controllers\Controller;
use App\Http\Resources\OrderResource;
use App\Models\Order;
use App\Services\OrderPaymentService;
use App\Services\StockService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class OrderController extends Controller
{
    public function index(Request $request): JsonResponse
    {
        $this->authorize('viewAny', Order::class);

        $orders = Order::query()
            ->where('user_id', $request->user()->id)
            ->with('items.itemable')
            ->latest('id')
            ->get();

        return response()->json([
            'data' => OrderResource::collection($orders),
        ]);
    }

    public function show(Request $request, Order $order): JsonResponse
    {
        $this->authorize('view', $order);

        $order->load('items.itemable', 'user', 'events');

        return response()->json([
            'data' => OrderResource::make($order),
        ]);
    }

    public function retryPayment(
        Request $request,
        OrderPaymentService $payments,
        StockService $stock,
    ): JsonResponse {
        if (! $request->user()) {
            return response()->json([
                'message' => __('api.order.unauthenticated'),
            ], 401);
        }

        $orderId = $request->route('order');

        if ($orderId instanceof Order) {
            $orderId = $orderId->getKey();
        }

        $order = Order::query()->findOrFail($orderId);

        if ((string) $order->user_id !== (string) $request->user()->getKey()) {
            return response()->json([
                'message' => __('api.order.not_yours'),
            ], 403);
        }

        if ($order->isPaid()) {
            return response()->json([
                'message' => __('api.order.already_paid'),
            ], 422);
        }

        if (! in_array($order->status, [OrderStatus::Pending, OrderStatus::Failed], true)) {
            return response()->json([
                'message' => __('api.order.cannot_retry'),
            ], 422);
        }

        try {
            DB::transaction(function () use ($order, $stock): void {
                $locked = Order::query()->whereKey($order->getKey())->lockForUpdate()->firstOrFail();

                if (! in_array($locked->status, [OrderStatus::Pending, OrderStatus::Failed], true)) {
                    throw new \RuntimeException(__('api.order.cannot_retry'));
                }

                $expiresAt = now()->addMinutes((int) config('shop.reservation_minutes', 30));

                if (! $locked->stock_reserved) {
                    // Réservation expirée ou paiement échoué : on reprend le stock, ou on refuse.
                    $stock->reserve($locked, $expiresAt);
                } else {
                    $locked->forceFill(['reservation_expires_at' => $expiresAt])->save();
                }

                if ($locked->status === OrderStatus::Failed) {
                    $locked->update(['status' => OrderStatus::Pending]);
                }
            });

            $order->refresh();

            if (config('payment.manual')) {
                return response()->json([
                    'payment_url' => null,
                    'transaction_id' => $order->payment_transaction_id,
                    'payment_pending_manual' => true,
                    'reservation_expires_at' => $order->reservation_expires_at,
                ]);
            }

            $payment = $payments->initiatePayment($order);
        } catch (InsufficientStockException $exception) {
            return response()->json($exception->toResponseData(), 422);
        } catch (\RuntimeException $exception) {
            return response()->json([
                'message' => $exception->getMessage(),
            ], 422);
        }

        return response()->json([
            'payment_url' => $payment['payment_url'],
            'transaction_id' => $payment['transaction_id'],
            'reservation_expires_at' => $order->reservation_expires_at,
        ]);
    }
}

    public function history(Order $order): JsonResponse
    {
        $this->authorize('manage', Order::class);

        $entries = [];
        $tx = $order->payment_transaction_id;

        if ($tx) {
            try {
                $events = PaymentWebhookEvent::query()
                    ->where(function (Builder $w) use ($tx): void {
                        $w->where('event_key', 'like', '%'.addcslashes((string) $tx, '%_\\').'%')
                            ->orWhere('payload->cpm_trans_id', $tx)
                            ->orWhere('payload->transaction_id', $tx);
                    })
                    ->latest('id')->limit(50)->get();
            } catch (\Throwable) {
                // Colonne payload non JSON : on se rabat sur la cle d'evenement seule
                $events = PaymentWebhookEvent::query()
                    ->where('event_key', 'like', '%'.addcslashes((string) $tx, '%_\\').'%')
                    ->latest('id')->limit(50)->get();
            }

            foreach ($events as $e) {
                $payload = is_array($e->payload) ? $e->payload : [];
                $parts = [];
                foreach (['cpm_result' => 'résultat', 'cpm_trans_status' => 'statut', 'cpm_amount' => 'montant', 'cpm_currency' => 'devise', 'payment_method' => 'moyen', 'cpm_error_message' => 'message'] as $key => $label) {
                    if (isset($payload[$key]) && $payload[$key] !== '') {
                        $parts[] = $label.' : '.(is_scalar($payload[$key]) ? $payload[$key] : json_encode($payload[$key]));
                    }
                }

                $entries[] = [
                    'id' => 'w'.$e->id,
                    'kind' => 'webhook',
                    'label' => 'Webhook reçu ('.$e->provider.')',
                    'at' => $this->iso($e->created_at),
                    'detail' => $parts === [] ? null : implode(' · ', $parts),
                ];
            }
        }

        $logs = ActivityLog::query()
            ->where('subject_type', $order->getMorphClass())
            ->where('subject_id', $order->getKey())
            ->whereIn('action', self::HISTORY_ACTIONS)
            ->latest('id')->limit(50)->get();

        foreach ($logs as $log) {
            $m = is_array($log->metadata) ? $log->metadata : [];
            $detail = $m['message'] ?? $m['reason'] ?? $m['note'] ?? $m['reference'] ?? $m['payment_reference'] ?? null;

            $entries[] = [
                'id' => 'a'.$log->id,
                'kind' => 'activity',
                'label' => self::ACTION_LABELS[$log->action] ?? $log->action,
                'at' => $this->iso($log->created_at),
                'detail' => is_scalar($detail) ? (string) $detail : null,
            ];
        }

        usort($entries, fn (array $a, array $b): int => strcmp((string) $b['at'], (string) $a['at']));

        return response()->json(['data' => $entries]);
    }

    public function verify(
        Request $request,
        Order $order,
        OrderFulfillmentService $fulfillment,
        ActivityLogger $activity,
        PaymentAnomalyService $anomalies,
    ): JsonResponse {
        $this->authorize('manage', Order::class);

        $result = $this->verifyOrder($order, $fulfillment, $activity, $anomalies, $request->user());

        return response()->json(['data' => $result + ['order' => $this->row($order->fresh('user'))]]);
    }

    public function verifyAll(
        Request $request,
        OrderFulfillmentService $fulfillment,
        ActivityLogger $activity,
        PaymentAnomalyService $anomalies,
    ): JsonResponse {
        $this->authorize('manage', Order::class);

        @set_time_limit(180);

        $base = Order::query()
            ->whereNotNull('payment_transaction_id')
            ->where(function (Builder $w): void {
                $w->where('status', 'pending')
                    ->orWhere(fn (Builder $x) => $x->where('status', 'failed')->whereNotNull('payment_claimed_at'));
            });

        $total = (clone $base)->count();

        // Les commandes jamais verifiees (ou verifiees le plus anciennement) passent en premier
        $orders = $base
            ->orderByRaw('payment_checked_at IS NOT NULL')
            ->orderBy('payment_checked_at')
            ->orderBy('id')
            ->limit(self::VERIFY_BATCH)
            ->get();

        $summary = [];
        foreach ($orders as $order) {
            $r = $this->verifyOrder($order, $fulfillment, $activity, $anomalies, $request->user());
            $summary[$r['result']] = ($summary[$r['result']] ?? 0) + 1;
        }

        return response()->json(['data' => [
            'checked' => $orders->count(),
            'remaining' => max(0, $total - $orders->count()),
            'summary' => $summary,
        ]]);
    }

    public function rejectClaim(Request $request, Order $order, ActivityLogger $activity): JsonResponse
    {
        $this->authorize('manage', Order::class);

        $data = $request->validate(['reason' => ['nullable', 'string', 'max:300']]);

        if ($order->isPaid()) {
            return response()->json(['message' => 'Cette commande est déjà payée.'], 422);
        }

        $previousReference = null;

        try {
            DB::transaction(function () use ($order, $data, &$previousReference): void {
                $locked = Order::query()->whereKey($order->getKey())->lockForUpdate()->firstOrFail();

                if ($locked->payment_claimed_at === null) {
                    throw new \RuntimeException('Aucune déclaration de paiement à rejeter pour cette commande.');
                }

                $previousReference = $locked->payment_claimed_reference;

                $locked->forceFill([
                    'payment_claimed_at' => null,
                    'payment_claimed_reference' => null,
                    'payment_claim_rejected_at' => now(),
                    'payment_claim_rejected_reason' => $data['reason'] ?? null,
                ])->save();
            });
        } catch (\RuntimeException $e) {
            return response()->json(['message' => $e->getMessage()], 422);
        }

        $activity->log($request->user(), 'order.payment_claim_rejected', $order, [
            'reason' => $data['reason'] ?? null,
            'reference' => $previousReference,
            'order_total' => $order->total,
        ]);

        return response()->json(['data' => $this->row($order->fresh('user'))]);
    }

    /** Suivi manuel : aucun appel a la passerelle, on note simplement que l'argent a ete rendu. */
    public function refund(Request $request, Order $order, ActivityLogger $activity): JsonResponse
    {
        $this->authorize('manage', Order::class);

        $data = $request->validate(['reason' => ['required', 'string', 'min:3', 'max:300']]);

        if (! $order->isPaid()) {
            return response()->json(['message' => 'Seule une commande payée peut être marquée remboursée.'], 422);
        }

        if ($order->getAttribute('refunded_at') !== null) {
            return response()->json(['message' => 'Cette commande est déjà marquée remboursée.'], 422);
        }

        $order->forceFill([
            'refunded_at' => now(),
            'refund_reason' => $data['reason'],
            'refunded_by' => $request->user()->getKey(),
        ])->save();

        $activity->log($request->user(), 'order.refunded', $order, [
            'reason' => $data['reason'],
            'order_total' => $order->total,
            'order_currency' => $order->currency,
        ]);

        return response()->json(['data' => $this->row($order->fresh('user'))]);
    }

    public function resolveAnomaly(Request $request, Order $order, PaymentAnomalyService $anomalies): JsonResponse
    {
        $this->authorize('manage', Order::class);

        $data = $request->validate(['note' => ['nullable', 'string', 'max:300']]);

        if ($order->getAttribute('payment_anomaly') === null) {
            return response()->json(['message' => 'Aucune anomalie à traiter sur cette commande.'], 422);
        }

        $anomalies->clear($order, $request->user(), $data['note'] ?? null);

        return response()->json(['data' => $this->row($order->fresh('user'))]);
    }

    public function export(Request $request): StreamedResponse
    {
        $this->authorize('manage', Order::class);

        $orders = $this->filteredQuery($request, $this->tab($request))->with('user')->latest('id')->limit(5000)->get();
        $filename = 'paiements-'.now()->format('Y-m-d').'.csv';

        // Neutralise les cellules interpretees comme formules par Excel
        $cell = function (mixed $v): string {
            $s = (string) ($v ?? '');

            return $s !== '' && in_array($s[0], ['=', '+', '-', '@', "\t", "\r"], true) ? "'".$s : $s;
        };

        return response()->streamDownload(function () use ($orders, $cell): void {
            $out = fopen('php://output', 'w');
            fwrite($out, "\xEF\xBB\xBF");
            $put = fn (array $row) => fputcsv($out, $row, ';', '"', '\\');

            $put(['ID', 'Référence', 'Date commande', 'Client', 'Email', 'Téléphone', 'Total', 'Devise', 'Statut',
                'Moyen de paiement', 'Référence paiement', 'Transaction', 'Déclaré le', 'Payé le',
                'Anomalie', 'Détail anomalie', 'Remboursé le', 'Motif remboursement']);

            foreach ($orders as $o) {
                $put([
                    $o->id,
                    $cell($o->public_ref),
                    $o->created_at?->format('Y-m-d H:i'),
                    $cell($o->user?->name ?? $o->guest_name),
                    $cell($o->user?->email ?? $o->guest_email),
                    $cell($o->phone),
                    $o->total,
                    $o->currency,
                    $o->status->value,
                    $cell($o->getAttribute('payment_method')),
                    $cell($o->getAttribute('payment_reference') ?? $o->payment_claimed_reference),
                    $cell($o->payment_transaction_id),
                    $this->iso($o->payment_claimed_at, 'Y-m-d H:i'),
                    $this->iso($o->paid_at, 'Y-m-d H:i'),
                    $cell($o->getAttribute('payment_anomaly') ?? ($o->stock_conflict ? 'stock_missing' : '')),

                    $this->iso($o->getAttribute('refunded_at'), 'Y-m-d H:i'),
                    $cell($o->getAttribute('refund_reason')),
                ]);
            }
            fclose($out);
        }, $filename, ['Content-Type' => 'text/csv; charset=UTF-8']);
    }

    /**
     * Relit le vrai statut chez CinetPay et applique le resultat.
     *
     * @return array{result: string, message: string}
     */
    private function verifyOrder(
        Order $order,
        OrderFulfillmentService $fulfillment,
        ActivityLogger $activity,
        PaymentAnomalyService $anomalies,
        ?User $actor,
    ): array {
        if ($order->isPaid()) {
            return ['result' => 'already_paid', 'message' => 'Cette commande est déjà payée.'];
        }

        $tx = $order->payment_transaction_id;
        if (! $tx) {
            return ['result' => 'no_transaction', 'message' => 'Aucune transaction CinetPay liée à cette commande (paiement manuel ?).'];
        }

        // Evite de reboucler sur la meme commande dans « Tout verifier », meme si la passerelle ne repond pas
        $order->forceFill(['payment_checked_at' => now()])->save();

        try {
            $res = $this->gateway()->checkTransactionStatus((string) $tx);
        } catch (\RuntimeException $e) {
            return ['result' => 'error', 'message' => 'Vérification impossible : '.$e->getMessage()];
        }

        $data = is_array($res['data'] ?? null) ? $res['data'] : [];
        $code = (string) ($res['code'] ?? '');
        $gwStatus = strtoupper((string) ($data['status'] ?? ''));

        if ($code === '00' && $gwStatus === 'ACCEPTED') {
            $paidAmount = (int) round((float) ($data['amount'] ?? 0));
            $expected = (int) $order->total;
            $currencyOk = strtoupper((string) ($data['currency'] ?? $order->currency)) === strtoupper((string) $order->currency);

            if ($paidAmount !== $expected || ! $currencyOk) {
                $note = "CinetPay a encaissé {$paidAmount} ".($data['currency'] ?? '').", la commande vaut {$expected} {$order->currency}.";
                $anomalies->flag($order, PaymentAnomalyService::AMOUNT_MISMATCH, $note, ['source' => 'verify'], $actor);

                return $this->logged($activity, $actor, $order, 'anomaly', 'Montant incohérent : '.$note, $gwStatus);
            }

            try {
                // Meme traitement que le webhook (OrderPaymentService)
                $fulfillment->markPaid($order, [
                    'method' => 'cinetpay',
                    'reference' => (string) $tx,
                    'amount' => $paidAmount,
                    'source' => 'gateway',
                    // pas de paid_by : paiement confirme par la passerelle (comme le webhook)
                ]);
            } catch (InsufficientStockException $e) {
                $anomalies->flag($order, PaymentAnomalyService::STOCK_MISSING, $e->getMessage(), ['source' => 'verify'], $actor);

                return $this->logged($activity, $actor, $order, 'anomaly', 'Paiement confirmé mais stock insuffisant : commande marquée en anomalie.', $gwStatus);
            } catch (\RuntimeException $e) {
                return ['result' => 'error', 'message' => 'Paiement confirmé chez CinetPay, mais impossible de l\'appliquer : '.$e->getMessage()];
            }

            $anomalies->clear($order->fresh(), $actor, 'Paiement confirmé par CinetPay');

            return $this->logged($activity, $actor, $order, 'paid', 'Paiement confirmé par CinetPay : commande marquée payée.', $gwStatus);
        }

        if ($code === '627') {
            return $this->logged($activity, $actor, $order, 'not_found', 'Transaction introuvable chez CinetPay.', $gwStatus ?: $code);
        }

        if (in_array($gwStatus, ['REFUSED', 'CANCELLED', 'FAILED'], true)) {
            // Meme traitement que le webhook (OrderPaymentService)
            if (method_exists($fulfillment, 'markFailed')) {
                try {
                    $fulfillment->markFailed($order);
                } catch (\Throwable $e) {
                    report($e);
                }
            }

            return $this->logged($activity, $actor, $order, 'refused', 'CinetPay indique que le paiement a été refusé.', $gwStatus);
        }

        return $this->logged($activity, $actor, $order, 'waiting', 'Le client n\'a pas encore finalisé le paiement chez CinetPay.', $gwStatus ?: $code);
    }

    /** @return array{result: string, message: string} */
    private function logged(ActivityLogger $activity, ?User $actor, Order $order, string $result, string $message, string $gateway): array
    {
        try {
            $activity->log($actor, 'order.payment_checked', $order, [
                'result' => $result,
                'gateway_status' => $gateway,
                'message' => $message,
            ]);
        } catch (\Throwable $e) {
            report($e);
        }

        return ['result' => $result, 'message' => $message];
    }

    private function gateway(): PaymentGateway
    {
        return app()->bound(PaymentGateway::class) ? app(PaymentGateway::class) : app(CinetPayService::class);
    }

    private function tab(Request $request): string
    {
        $tab = (string) $request->query('tab', 'to_verify');

        return in_array($tab, self::TABS, true) ? $tab : 'all';
    }

    private function scopeTab(Builder $q, string $tab): Builder
    {
        $now = now();

        return match ($tab) {
            'to_verify' => $q->whereIn('status', ['pending', 'failed'])->whereNotNull('payment_claimed_at'),
            'pending' => $q->where('status', 'pending')->whereNull('payment_claimed_at')
                ->where(fn (Builder $w) => $w->whereNull('reservation_expires_at')->orWhere('reservation_expires_at', '>', $now)),
            'failed' => $q->whereNull('payment_claimed_at')
                ->where(fn (Builder $w) => $w->where('status', 'failed')
                    ->orWhere(fn (Builder $x) => $x->where('status', 'pending')
                        ->whereNotNull('reservation_expires_at')->where('reservation_expires_at', '<=', $now))),
            'anomalies' => $q->where(fn (Builder $w) => $w->where('stock_conflict', true)->orWhereNotNull('payment_anomaly')),
            'paid_today' => $q->where('status', 'paid')->where('paid_at', '>=', $now->copy()->startOfDay()),
            default => $q,
        };
    }

    private function filteredQuery(Request $request, string $tab): Builder
    {
        $q = trim((string) $request->query('q', ''));
        $from = (string) $request->query('from', '');
        $to = (string) $request->query('to', '');

        return $this->scopeTab(Order::query(), $tab)
            ->when(preg_match('/^\d{4}-\d{2}-\d{2}$/', $from) === 1, fn ($w) => $w->whereRaw('COALESCE(paid_at, created_at) >= ?', [$from.' 00:00:00']))
            ->when(preg_match('/^\d{4}-\d{2}-\d{2}$/', $to) === 1, fn ($w) => $w->whereRaw('COALESCE(paid_at, created_at) <= ?', [$to.' 23:59:59']))
            ->when($q !== '', function ($w) use ($q): void {
                $like = '%'.mb_strtolower(addcslashes($q, '%_\\')).'%';
                $id = ltrim($q, '#');
                $w->where(function ($x) use ($like, $id): void {
                    if (ctype_digit($id)) {
                        $x->orWhere('id', (int) $id);
                    }
                    $x->orWhereRaw('LOWER(public_ref) LIKE ?', [$like])
                        ->orWhereRaw('LOWER(guest_name) LIKE ?', [$like])
                        ->orWhereRaw('LOWER(guest_email) LIKE ?', [$like])
                        ->orWhereRaw('LOWER(phone) LIKE ?', [$like])
                        ->orWhereRaw('LOWER(payment_transaction_id) LIKE ?', [$like])
                        ->orWhereRaw('LOWER(payment_claimed_reference) LIKE ?', [$like])
                        ->orWhereHas('user', fn ($u) => $u
                            ->whereRaw('LOWER(name) LIKE ?', [$like])
                            ->orWhereRaw('LOWER(email) LIKE ?', [$like]));
                });
            });
    }

    /** @return array<string, int> */
    private function counts(): array
    {
        $counts = [];
        foreach (['to_verify', 'pending', 'failed', 'anomalies', 'paid_today'] as $tab) {
            $counts[$tab] = $this->scopeTab(Order::query(), $tab)->count();
        }

        $counts['collected_today'] = (int) $this->scopeTab(Order::query(), 'paid_today')->sum('total');

        return $counts;
    }

    /** @return array<string, mixed> */
    private function row(Order $o): array
    {
        $status = $o->status->value;
        $claimed = $o->payment_claimed_at !== null;
        $expired = $status === 'pending' && $o->reservation_expires_at !== null && $o->reservation_expires_at->lte(now());

        $bucket = match (true) {
            $status === 'paid' => 'paid',
            $status === 'cancelled' => 'cancelled',
            $claimed => 'to_verify',
            $status === 'failed' || $expired => 'failed',
            default => 'pending',
        };

        return [
            'id' => $o->id,
            'public_ref' => $o->public_ref,
            'customer' => $o->user?->name ?? $o->guest_name,
            'email' => $o->user?->email ?? $o->guest_email,
            'phone' => $o->phone,
            'total' => (int) $o->total,
            'currency' => $o->currency,
            'status' => $status,
            'bucket' => $bucket,
            'created_at' => $this->iso($o->created_at),
            'paid_at' => $this->iso($o->paid_at),
            'payment_method' => $o->getAttribute('payment_method'),
            'payment_reference' => $o->getAttribute('payment_reference'),
            'payment_transaction_id' => $o->payment_transaction_id,
            'claimed_at' => $this->iso($o->payment_claimed_at),
            'claimed_reference' => $o->payment_claimed_reference,
            'claim_rejected_at' => $this->iso($o->getAttribute('payment_claim_rejected_at')),
            'claim_rejected_reason' => $o->getAttribute('payment_claim_rejected_reason'),
            'checked_at' => $this->iso($o->getAttribute('payment_checked_at')),
            'stock_conflict' => (bool) $o->stock_conflict,
            'anomaly' => $o->getAttribute('payment_anomaly'),
            'anomaly_note' => $o->getAttribute('payment_anomaly_note'),
            'refunded_at' => $this->iso($o->getAttribute('refunded_at')),
            'refund_reason' => $o->getAttribute('refund_reason'),
            'reservation_expires_at' => $this->iso($o->reservation_expires_at),
        ];
    }

    private function iso(mixed $value, ?string $format = null): ?string
    {
        if ($value === null || $value === '') {
            return null;
        }

        $date = $value instanceof Carbon ? $value : Carbon::parse($value);

        return $format ? $date->format($format) : $date->toIso8601String();
    }
}

export const apiClient = axios.create({
  baseURL: "/api/v1",
  timeout: 20_000,
  withCredentials: true,
  withXSRFToken: true,

apiClient.interceptors.request.use((config) => {
  const token = localStorage.getItem(TOKEN_KEY)
  if (token) config.headers.Authorization = `Bearer ${token}`
  config.headers["Accept-Language"] = i18n.language.slice(0, 2)
  return config
})

apiClient.interceptors.response.use(
  (response) => response,
  async (error) => {
    if (axios.isAxiosError(error) && error.response) {
      // Téléchargements (blob) : on récupère le message JSON de Laravel
      const { data } = error.response
      if (data instanceof Blob && data.type.includes("json")) {
        try {
          error.response.data = JSON.parse(await data.text())
        } catch {
          // on garde le blob tel quel
        }
      }
      if (error.response.status === 401 && localStorage.getItem(TOKEN_KEY)) {
        localStorage.removeItem(TOKEN_KEY)
        window.dispatchEvent(new Event("auth:expired"))
      }
    }
    return Promise.reject(error)
  },
)

import axios from "axios"
import i18n from "../i18n"
import type { ApiErrorBody } from "../types"

export type ErrorKind =
  | "offline" | "timeout" | "network" | "unauthorized" | "forbidden" | "not_found"
  | "expired" | "validation" | "rate_limit" | "server" | "unknown"

export interface ErrorInfo {
  kind: ErrorKind
  status?: number
  message: string
  retryable: boolean
}

const tr = (key: string, fallback: string) => i18n.t(key, { defaultValue: fallback })

export function getErrorInfo(error: unknown): ErrorInfo {
  if (!axios.isAxiosError<ApiErrorBody>(error)) {
    return { kind: "unknown", message: tr("errors.unexpected", "Une erreur inattendue est survenue."), retryable: false }
  }

  const res = error.response
  if (!res) {
    if (!navigator.onLine) return { kind: "offline", message: tr("errors.offline", "Vous êtes hors ligne. Vérifiez votre connexion."), retryable: true }
    if (error.code === "ECONNABORTED" || error.code === "ETIMEDOUT") return { kind: "timeout", message: tr("errors.timeout", "Le serveur met trop de temps à répondre."), retryable: true }
    return { kind: "network", message: tr("errors.network", "Impossible de joindre le serveur."), retryable: true }
  }

  const { status, data } = res
  const firstField = data?.errors ? Object.values(data.errors)[0]?.[0] : undefined
  const serverMessage = firstField ?? data?.message

  if (status === 401) return { kind: "unauthorized", status, message: tr("errors.session_expired", "Votre session a expiré. Reconnectez-vous."), retryable: false }
  if (status === 403) return { kind: "forbidden", status, message: serverMessage ?? tr("errors.forbidden", "Vous n'avez pas accès à cette ressource."), retryable: false }
  if (status === 404) return { kind: "not_found", status, message: tr("errors.not_found", "Cet élément est introuvable."), retryable: false }
  if (status === 419) return { kind: "expired", status, message: tr("errors.csrf", "La page a expiré. Rechargez-la puis réessayez."), retryable: false }
  if (status === 422) return { kind: "validation", status, message: serverMessage ?? tr("errors.validation", "Certaines informations sont invalides."), retryable: false }
  if (status === 429) return { kind: "rate_limit", status, message: tr("errors.rate_limit", "Trop de tentatives. Patientez un instant."), retryable: true }
  if (status >= 500) return { kind: "server", status, message: tr("errors.server", "Un problème est survenu de notre côté. Réessayez."), retryable: true }
  return { kind: "unknown", status, message: serverMessage ?? tr("errors.unexpected", "Une erreur inattendue est survenue."), retryable: false }
}

export const getErrorMessage = (error: unknown) => getErrorInfo(error).message

  if (isLoading) return <ProductDetailSkeleton />
  if (error || !data) return <ErrorState error={error} onRetry={() => void refetch()} />

  const soldOut = data.in_stock === false
  const gallery = data.images && data.images.length > 0 ? data.images : data.image_url ? [data.image_url] : []
  const max = typeof data.low_stock === "number" ? data.low_stock : 99
  const en = i18n.language.startsWith("en")
  const extra = data as unknown as { specs?: string | null; delivery_info?: string | null }
  const quantity = Math.min(qty, max)
  const url = data.share_url ?? `${window.location.origin}/boutique/${data.slug}`

  // Meme rangee dans la barre collee et dans le plein ecran : prix, ajout, WhatsApp carre
  const buyRow = (
    <div className="flex items-center gap-3">
      <p className="text-xl font-extrabold text-primary">{formatPrice(data.price * quantity)}</p>
      <div className="min-w-0 flex-1">
        <AddToCartButton type="product" id={data.id} quantity={quantity} inStock={data.in_stock} />
      </div>
      <WhatsAppButton productName={data.name} productUrl={url} soldOut={soldOut} display="icon" />
    </div>
  )

  return (
    <div className="pb-24 lg:pb-0">
      <BackLink to="/boutique">{t("product.back")}</BackLink>
      <div className="mt-2 grid gap-6 lg:mt-4 lg:grid-cols-3 lg:gap-8">
        <div className="lg:col-span-2">
          <Gallery images={gallery} thumbs={(data as unknown as { thumbs?: string[] }).thumbs} alt={data.name} footer={buyRow} />
          <h1 className="mt-6 text-2xl font-bold sm:text-3xl">{data.name}</h1>
          <div className="mt-2 flex flex-wrap items-center gap-3 lg:hidden">
            <p className="text-2xl font-extrabold text-primary">{formatPrice(data.price)}</p>
            <StockBadge inStock={data.in_stock} lowStock={data.low_stock} />
          </div>
          {!soldOut && (
            <div className="mt-4 lg:hidden">
              <QuantityStepper value={quantity} onChange={setQty} max={max} />
            </div>
          )}
          <div className="mt-3 lg:hidden">
            <WhatsAppButton productName={data.name} productUrl={url} soldOut={soldOut} display="link" className="!justify-start !px-0" />
          </div>
          <h2 data-tour="pd-desc" className="mt-6 text-lg font-semibold">{t("product.description")}</h2>
          <div className="mt-3">
            {data.description ? <RichText text={data.description} /> : <p className="text-muted">{t("product.no_description")}</p>}
          </div>
          {extra.specs && (
            <section className="mt-8">
              <h2 className="mb-3 text-lg font-semibold">{en ? "Specifications" : "Caractéristiques"}</h2>
              <SpecsTable text={extra.specs} />
            </section>
          )}
          {extra.delivery_info && (
            <section className="mt-8 rounded-card border border-line bg-surface p-4 shadow-card sm:p-5">
              <h2 className="mb-3 text-lg font-semibold">{en ? "Delivery and returns" : "Livraison et retours"}</h2>
              <RichText text={extra.delivery_info} />
            </section>
          )}
          <div className="mt-10"><FaqAccordion space="shop" compact productId={data.id} /></div>
        </div>

        <aside data-tour="pd-buy" className="sticky top-24 hidden h-fit rounded-card border border-line bg-surface p-5 shadow-card lg:block">
          <p className="text-3xl font-extrabold text-primary">{formatPrice(data.price)}</p>
          <StockBadge inStock={data.in_stock} lowStock={data.low_stock} className="mt-2" />
          {!soldOut && (
            <div className="mt-4">
              <QuantityStepper value={quantity} onChange={setQty} max={max} />
            </div>
          )}
          <div className="mt-4">
            <AddToCartButton type="product" id={data.id} quantity={quantity} inStock={data.in_stock} />
          </div>
          <div className="mt-2">
            <WhatsAppButton productName={data.name} productUrl={url} soldOut={soldOut} display="link" />
          </div>
        </aside>
      </div>

      <div className="fixed inset-x-0 bottom-[calc(4rem+env(safe-area-inset-bottom))] z-30 border-t border-line bg-surface/95 px-4 pb-3 pt-3 backdrop-blur lg:hidden">
        <div data-tour="pd-buy-bar" className="mx-auto max-w-6xl">{buyRow}</div>
      </div>
      <div className="mt-10"><Suggestions excludeIds={[data.id]} /></div>
      <div className="mt-10"><RecentlyViewed excludeSlug={data.slug} /></div>
    </div>
  )
}






===== .\frontend\src\pages\ProductsPage.tsx =====
import { useEffect, useMemo, useRef, useState } from "react"
import { keepPreviousData, useInfiniteQuery, useQuery } from "@tanstack/react-query"
import { useSearchParams } from "react-router-dom"
import { useTranslation } from "react-i18next"
import { catalogApi } from "../api/catalog"
import ProductCard from "../components/ProductCard"
import Button from "../components/Button"
import ShopFilterSheet, { type ShopFilters } from "../components/ShopFilterSheet"
import { EmptyState, ErrorState } from "../components/States"
import { ProductGridSkeleton } from "../components/Skeleton"
import RecentlyViewed from "../components/RecentlyViewed"
import type { Product } from "../types"

const SORTS = [
  { value: "new", key: "ux.sort_new" },
  { value: "price_asc", key: "ux.sort_price_asc" },
  { value: "price_desc", key: "ux.sort_price_desc" },
]


const field =
  "min-h-[48px] rounded-lg border border-line bg-surface px-3.5 py-2.5 outline-none transition focus:border-primary focus:ring-2 focus:ring-primary/20"

const chip = (active: boolean) =>
  `min-h-[40px] flex-none rounded-full border px-4 text-sm font-medium transition ${
    active ? "border-primary bg-primary text-white" : "border-line bg-surface text-ink hover:border-primary/50"
  }`

export default function ProductsPage() {
  const { t, i18n } = useTranslation()
  const fr = i18n.language.startsWith("fr")
  const [params, setParams] = useSearchParams()
  const q = params.get("q") ?? ""
  const sort = params.get("sort") ?? "new"
  const onlyStock = params.get("stock") === "1"
  const onlyNew = params.get("new") === "1"
  const category = params.get("cat") ?? ""
  const minPrice = Number(params.get("min")) || 0
  const maxPrice = Number(params.get("max")) || 0
  const [input, setInput] = useState(q)
  const [sheet, setSheet] = useState(false)
  const cats = useQuery({ queryKey: ["categories"], queryFn: async () => (await catalogApi.categories()).data.data, staleTime: 60_000 })

  function update(next: Record<string, string>) {
    const p = new URLSearchParams(params)
    for (const [k, v] of Object.entries(next)) {
      if (v) p.set(k, v)
      else p.delete(k)
    }
    p.delete("page")
    setParams(p, { replace: true })
  }

  useEffect(() => {
    if (input.trim() === q) return
    const id = setTimeout(() => update({ q: input.trim() }), 300)
    return () => clearTimeout(id)
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [input])

  const { data, isLoading, isPlaceholderData, error, refetch, fetchNextPage, hasNextPage, isFetchingNextPage } = useInfiniteQuery({
    queryKey: ["products-inf", { q, sort, category, minPrice, maxPrice, onlyStock, onlyNew }],
    queryFn: async ({ pageParam }) => (await catalogApi.products({ page: pageParam, search: q || undefined, sort, category: category || undefined, min_price: minPrice || undefined, max_price: maxPrice || undefined, in_stock: onlyStock ? 1 : undefined, new: onlyNew ? 1 : undefined })).data,
    initialPageParam: 1,
    getNextPageParam: (last) => (last.meta.current_page < last.meta.last_page ? last.meta.current_page + 1 : undefined),
    placeholderData: keepPreviousData,
  })

  const all = useMemo(() => {
    const seen = new Set<number>()
    const out: Product[] = []
    for (const page of data?.pages ?? []) {
      for (const p of page.data) {
        if (!seen.has(p.id)) {
          seen.add(p.id)
          out.push(p)
        }
      }
    }
    return out
  }, [data])

  const shown = all
  const filtering = onlyStock || onlyNew || !!category || !!minPrice || !!maxPrice
  const total = data?.pages[0]?.meta.total ?? 0
  const activeCount = (onlyStock ? 1 : 0) + (onlyNew ? 1 : 0) + (category ? 1 : 0) + (minPrice || maxPrice ? 1 : 0)

  const clearFilters = () => update({ stock: "", new: "", cat: "", min: "", max: "" })

  // Pastilles des filtres actifs
  const fmt = (n: number) => n.toLocaleString(fr ? "fr-FR" : "en-US")
  const priceLabel =
    minPrice && maxPrice ? `${fmt(minPrice)} – ${fmt(maxPrice)} FCFA`
    : minPrice ? `≥ ${fmt(minPrice)} FCFA`
    : maxPrice ? `≤ ${fmt(maxPrice)} FCFA`
    : ""
  const catName = cats.data?.find((c) => c.slug === category)?.name ?? category
  const pills: { key: string; label: string; remove: () => void }[] = []
  if (q) pills.push({ key: "q", label: `« ${q} »`, remove: () => { setInput(""); update({ q: "" }) } })
  if (category) pills.push({ key: "cat", label: catName, remove: () => update({ cat: "" }) })
  if (onlyStock) pills.push({ key: "stock", label: fr ? "En stock" : "In stock", remove: () => update({ stock: "" }) })
  if (onlyNew) pills.push({ key: "new", label: t("ux.new_badge"), remove: () => update({ new: "" }) })
  if (priceLabel) pills.push({ key: "price", label: priceLabel, remove: () => update({ min: "", max: "" }) })

  // Charge la suite automatiquement quand on approche du bas de la liste
  const sentinel = useRef<HTMLDivElement>(null)
  useEffect(() => {
    const el = sentinel.current
    if (!el || !hasNextPage || isPlaceholderData) return
    const io = new IntersectionObserver(
      (entries) => {
        if (entries[0]?.isIntersecting && !isFetchingNextPage) void fetchNextPage()
      },
      { rootMargin: "400px" },
    )
    io.observe(el)
    return () => io.disconnect()
  }, [hasNextPage, isFetchingNextPage, isPlaceholderData, fetchNextPage, shown.length])

  return (
    <div>
      <h1 className="mb-4 text-2xl font-bold sm:text-3xl">{t("shop.title")}</h1>

      <div className="mb-5 space-y-3">
        <div className="relative">
          <svg
            className="pointer-events-none absolute left-3.5 top-1/2 h-5 w-5 -translate-y-1/2 text-muted"
            viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true"
          >
            <circle cx="11" cy="11" r="7" />
            <path d="M21 21l-4.3-4.3" />
          </svg>
          <input
            type="text" inputMode="search" data-tour="search" value={input}
            onChange={(e) => setInput(e.target.value)}
            placeholder={t("ux.search")} aria-label={t("ux.search")} enterKeyHint="search"
            className={`${field} w-full pl-11 ${input ? "pr-11" : ""}`}
          />
          {input && (
            <button
              type="button" onClick={() => { setInput(""); update({ q: "" }) }}
              aria-label={t("ux.clear", { defaultValue: "Effacer" })}
              className="absolute inset-y-0 right-0 flex w-11 items-center justify-center text-muted hover:text-ink"
            >
              <svg className="h-4 w-4" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" aria-hidden="true"><path d="M6 6l12 12M18 6L6 18" /></svg>
            </button>
          )}
        </div>

        <div className="flex gap-2" data-tour="filters">
          <button
            type="button" onClick={() => setSheet(true)} aria-haspopup="dialog"
            className={`${field} flex shrink-0 items-center gap-2 font-semibold hover:border-primary/50`}
          >
            <svg className="h-[18px] w-[18px]" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
              <path d="M4 6h10M18 6h2M4 12h4M12 12h8M4 18h12M20 18h0" /><circle cx="16" cy="6" r="2" /><circle cx="10" cy="12" r="2" /><circle cx="18" cy="18" r="2" />
            </svg>
            {fr ? "Filtres" : "Filters"}
            {activeCount > 0 && <span className="rounded-full bg-primary px-2 text-xs font-bold text-white">{activeCount}</span>}
          </button>
          <select
            value={sort} onChange={(e) => update({ sort: e.target.value === "new" ? "" : e.target.value })}
            aria-label={t("ux.sort")} className={`${field} min-w-0 flex-1`}
          >
            {SORTS.map((s) => <option key={s.value} value={s.value}>{t(s.key)}</option>)}
          </select>
        </div>

        {cats.data && cats.data.length > 0 && (
          <div className="-mx-4 flex gap-2 overflow-x-auto px-4 [scrollbar-width:none] sm:mx-0 sm:px-0" role="group" aria-label={fr ? "Catégories" : "Categories"}>
            <button type="button" aria-pressed={category === ""} onClick={() => update({ cat: "" })} className={chip(category === "")}>
              {fr ? "Toutes" : "All"}
            </button>
            {cats.data.map((c) => (
              <button key={c.id} type="button" aria-pressed={category === c.slug} onClick={() => update({ cat: c.slug })} className={chip(category === c.slug)}>
                {c.name}
              </button>
            ))}
          </div>
        )}

        {pills.length > 0 && (
          <div className="flex flex-wrap items-center gap-2" aria-label={fr ? "Filtres actifs" : "Active filters"}>
            {pills.map((p) => (
              <span key={p.key} className="inline-flex min-h-[32px] items-center gap-1 rounded-full bg-primary/10 pl-3 pr-1 text-xs font-semibold text-primary">
                {p.label}
                <button
                  type="button" onClick={p.remove} aria-label={`${fr ? "Retirer" : "Remove"} ${p.label}`}
                  className="flex h-6 w-6 items-center justify-center rounded-full text-sm leading-none hover:bg-primary/15"
                >
                  &times;
                </button>
              </span>
            ))}
            {pills.length > 1 && (
              <button type="button" onClick={() => { setInput(""); update({ q: "" }); clearFilters() }} className="px-1 text-xs font-semibold text-muted underline hover:text-danger">
                {fr ? "Tout effacer" : "Clear all"}
              </button>
            )}
          </div>
        )}

        {data && all.length > 0 && (
          <p className="text-sm text-muted" aria-live="polite">
            {filtering
              ? fr ? `${total} produit(s)` : `${total} product(s)`
              : t("shop.count", { count: total, defaultValue: "{{count}} produit(s)" })}
          </p>
        )}
      </div>

      {sheet && (
        <ShopFilterSheet
          fr={fr}
          categories={cats.data ?? []}
          initial={{ category, stock: onlyStock, fresh: onlyNew, min: minPrice ? String(minPrice) : "", max: maxPrice ? String(maxPrice) : "" } satisfies ShopFilters}
          onClose={() => setSheet(false)}
          onApply={(f) => {
            update({ cat: f.category, stock: f.stock ? "1" : "", new: f.fresh ? "1" : "", min: f.min, max: f.max })
            setSheet(false)
          }}
        />
      )}

      {isLoading && <ProductGridSkeleton />}
      {error && !data && <ErrorState error={error} onRetry={() => void refetch()} />}

      {data && shown.length === 0 && !hasNextPage && (
        <EmptyState
          message={q ? t("ux.no_results", { q }) : filtering ? (fr ? "Aucun produit ne correspond à ces filtres" : "No product matches these filters") : t("shop.empty")}
          actionLabel={filtering ? (fr ? "Retirer les filtres" : "Clear filters") : undefined}
          onAction={filtering ? clearFilters : undefined}
        />
      )}

      {shown.length > 0 && (
        <div className={`stagger grid grid-cols-2 gap-3 transition-opacity sm:gap-4 md:grid-cols-3 lg:grid-cols-4 ${isPlaceholderData ? "opacity-60" : ""}`}>
          {shown.map((product) => (
            <ProductCard key={product.id} product={product} />
          ))}
        </div>
      )}

      {hasNextPage && (
        <div ref={sentinel} className="mt-6 flex justify-center">
          <Button variant="secondary" loading={isFetchingNextPage} onClick={() => void fetchNextPage()}>
            {fr ? "Voir plus de produits" : "Show more products"}
          </Button>
        </div>
      )}
      {!hasNextPage && shown.length > 0 && (
        <p className="mt-6 text-center text-sm text-muted">{fr ? "Vous avez tout vu." : "You've seen everything."}</p>
      )}

      {!q && <div className="mt-10"><RecentlyViewed /></div>}
    </div>
  )
}
